   36 def weapon_dice(rec):
   37     """(n, s) of a weapon's damage: high nibble dice, low nibble sides."""
   38     if rec is None:
   39         return 1, 6
   40     if "damage_n" in rec:
   41         return rec.get("damage_n", 1), rec.get("damage_s", 6)
   42     byte = rec.get("damage", 0) or 0
   43     n = (byte >> 4) & 0x0F
   44     s = byte & 0x0F
   45     if n <= 0:
   46         n = 1
   47     if s <= 0:
   48         s = 6
   49     return n, s
