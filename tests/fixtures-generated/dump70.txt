  13 SKILL_ENTRIES = 36
  14 MASK32 = 0xFFFFFFFF
  15 DEFAULT_SEED = 0x6D2B79F5

  18 def read_skill_table(path: str) -> list:
  19     """36 x ``[minimum_iq, cost, attribute_offset]`` (SPEC-04 LR-6)."""
  20     with open(path, "rb") as fh:
  21         data = fh.read()
  22     if len(data) < 2 * SKILL_ENTRIES:
  23         raise FormatError("LABSKILL: %d bytes, expected at least %d"
  24                           % (len(data), 2 * SKILL_ENTRIES))
  25     out = []
  26     for i in range(SKILL_ENTRIES):
  27         b0, b1 = data[2 * i], data[2 * i + 1]
  28         out.append([b0 >> 3, b0 & 7, b1])
  29     return out

  36 class Rng:
  37     """xorshift32 generator of SPEC-04 LR-1 and LR-2."""
  38 
  39     def __init__(self, seed: int = 0) -> None:
  40         seed = int(seed) & MASK32
  41         self.state = seed if seed else DEFAULT_SEED
  42 
  43     def next(self) -> int:
  44         x = self.state
  45         x ^= (x << 13) & MASK32
  46         x ^= x >> 17
  47         x ^= (x << 5) & MASK32
  48         x &= MASK32
  49         self.state = x
  50         return x
  51 
  52     def roll(self, n: int) -> int:
  53         return 1 + self.next() % n
  54 
  55     def rnd(self, n: int) -> int:
  56         if n == 0:
  57             return 0
  58         if n == 1:
  59             return 1
  60         return self.roll(n)
  61 
  62     def dice(self, n: int, sides: int) -> int:
  63         total = 0
  64         for _ in range(n):
  65             total += self.roll(sides)
  66         return total
  67 
  68     def d100(self) -> int:
  69         return self.roll(100)

 100 def check_need(level: int, attribute: int, difficulty: int, tool: bool = False) -> int:
 101     """LR-13, unclamped."""
 102     return 20 + 15 * level + 2 * attribute - difficulty + (20 if tool else 0)

 105 def skill_check(rng: Rng, level: int, attribute: int, difficulty: int, tool: bool = False) -> dict:
 106     """LR-13: one d100 unless the need is 100 or more, or 0 or less."""
 107     need = check_need(level, attribute, difficulty, tool)
 108     if need >= 100:
 109         return {"need": need, "roll": None, "success": True}
 110     if need <= 0:
 111         return {"need": need, "roll": None, "success": False}
 112     r = rng.d100()
 113     return {"need": need, "roll": r, "success": r <= need}

 116 def pick_checker(candidates: list):
 117     """LR-14: highest level, then higher attribute, then lower slot."""
 118     best = None
 119     best_key = None
 120     for c in candidates:
 121         if c.get("cond", 0) != 0:
 122             continue
 123         key = (c.get("level", 0), c.get("attribute", 0), -c.get("slot", 0))
 124         if best_key is None or key > best_key:
 125             best_key = key
 126             best = c.get("slot")
 127     return best

 134 def advance_clock(clock: dict, minutes: int, seconds: int):
 135     """LR-16, LR-17: returns (clock, hours whose minute 0 was reached)."""
 136     day = clock.get("day", 1)
 137     minute = clock.get("minute", 0)
 138     second = clock.get("second", 0)
 139     total_s = second + seconds
 140     second = total_s % 60
 141     add_min = minutes + total_s // 60
 142     hours = []
 143     for _ in range(add_min):
 144         minute += 1
 145         if minute == 1440:
 146             minute = 0
 147         if minute % 60 == 0:
 148             h = minute // 60
 149             hours.append(h)
 150             if h == 0:
 151                 day += 1
 152     return {"day": day, "minute": minute, "second": second}, hours

 187 def pay(moneys: list, first_slot: int, price: int):
 188     """LR-20: new money list, or None when the party cannot pay."""
 189     if sum(moneys) < price:
 190         return None
 191     out = list(moneys)
 192     order = [first_slot - 1] + [i for i in range(len(moneys)) if i != first_slot - 1]
 193     left = price
 194     for i in order:
 195         take = min(out[i], left)
 196         out[i] -= take
 197         left -= take
 198         if left == 0:
 199             break
 200     return out

 207 def monster_values(mob: dict) -> dict:
 208     """LR-27."""
 209     cr = mob.get("combatRating", 0)
 210     ac = mob.get("armorClass", 0)
 211     st = mob.get("sturdiness", 0)
 212     return {"to_hit_base": 20 + 5 * cr, "speed": cr, "ac": ac,
 213             "ranged": mob.get("weaponType", 0) >= 3, "xp": st * (ac + 1)}

 216 def monster_hp(rng: Rng, sturdiness: int) -> int:
 217     """SPEC-03 6.6, LR-26: the high byte is drawn first."""
 218     hi = rng.rnd(sturdiness >> 8)
 219     lo = rng.rnd(sturdiness & 0xFF)
 220     return sturdiness // 4 + ((hi << 8) | lo)
 221 
 222 
 223 def monster_damage(rng: Rng, mob: dict) -> int:
 224     """LR-27."""
 225     return rng.dice(mob.get("randomDamage", 0), 6) + mob.get("fixedDamage", 0)
 226 
 227 
 228 def build_groups(rng: Rng, mobs: list, record: dict) -> list:
 229     """LR-26, LR-28."""
 230     counts = []
 231     for entry in record.get("groups", []):
 232         mob_index, rnd_flag, size = entry[0], entry[1], entry[2]
 233         if mob_index == 0:
 234             continue
 235         mob = mobs[mob_index - 1]
 236         s = size if size else mob.get("maxGroup", 0)
 237         counts.append((mob_index, rng.rnd(s) if rnd_flag else s))
 238     out = []
 239     for mob_index, count in counts:
 240         if count <= 0:
 241             continue
 242         mob = mobs[mob_index - 1]
 243         hp = [monster_hp(rng, mob.get("sturdiness", 0)) for _ in range(count)]
 244         band = 3 if mob.get("weaponType", 0) >= 3 else 2
 245         out.append({"monster": mob_index, "count": count, "band": band, "hp": hp})
 246     return out

 292 def damage_member(hp: int, damage: int) -> dict:
 293     """LR-40."""
 294     prev = hp
 295     hp -= damage
 296     if hp > 0:
 297         return {"hp": hp, "cond": 0, "last_hp": None}
 298     over = damage - prev
 299     return {"hp": 0, "cond": 2 if over >= 5 else 1, "last_hp": prev}

 304     def key(a):

 308                 0 if party else a.get("group", 0),
 309                 0 if party else a.get("index", 0))

  71 def first_diff(got, want, path: str = ""):
  72     """Return (path, got_part, want_part) of the first difference, or None."""
  73     if isinstance(want, dict):
  74         if not isinstance(got, dict):
  75             return (path or "<root>", got, "a dict")

  60 class RefRng:
  61     """SPEC-04 LR-1 and LR-2, restated for computing expected values."""

  63     def __init__(self, seed: int) -> None:
  64         self.state = seed & 0xFFFFFFFF or 0x6D2B79F5
  65 
  66     def next(self) -> int:
  67         x = self.state
  68         x ^= (x << 13) & 0xFFFFFFFF
  69         x ^= x >> 17
  70         x ^= (x << 5) & 0xFFFFFFFF
  71         self.state = x
  72         return x

  74     def roll(self, n: int) -> int:
  75         return 1 + self.next() % n

  77     def rnd(self, n: int) -> int:
  78         if n == 0:

  80         if n == 1:
  81             return 1
  82         return self.roll(n)
  83 
  84     def dice(self, n: int, s: int) -> int:
  85         return sum(self.roll(s) for _ in range(n))

  87     def d100(self) -> int:

 172 def ref_pay(moneys: list[int], first_slot: int, price: int):
 173     if sum(moneys) < price:
 174         return None
 175     out = list(moneys)
 176     order = [first_slot - 1] + [i for i in range(len(moneys)) if i != first_slot - 1]
 177     left = price
 178     for i in order:
 179         take = min(out[i], left)
 180         out[i] -= take
 181         left -= take
 182         if left == 0:
 183             break
 184     return out

 187 def ref_advance_clock(clock: dict, minutes: int, seconds: int):
 188     day, minute, second = clock["day"], clock["minute"], clock["second"]
 189     total_s = second + seconds
 190     second = total_s % 60
 191     add_min = minutes + total_s // 60
 192     hours = []
 193     for _ in range(add_min):
 194         minute += 1
 195         if minute == 1440:

 233 def ref_labstate(blob: bytes) -> dict:
 234     """Independent LABSTATE.BIN reader (SPEC-04 LR-57)."""
 235     if blob[:8] != b"WLLABSV1":
 236         raise ValueError("bad magic")
 237     n = int.from_bytes(blob[0x14:0x16], "little")
 238     end = 0x16 + 6 * n
 239     if len(blob) != end + 2:
 240         raise ValueError("bad length")
 241     if sum(blob[:end]) & 0xFFFF != int.from_bytes(blob[end:end + 2], "little"):
 242         raise ValueError("bad sum")
 243     cc = [list(blob[0x16 + 6 * i:0x16 + 6 * i + 6]) for i in range(n)]
 244     return {"steps": int.from_bytes(blob[8:12], "little"), "day": int.from_bytes(blob[12:14], "little"),
 245             "seconds": blob[14], "pad": blob[15], "rng_state": int.from_bytes(blob[16:20], "little"),
 246             "cell_changes": cc}

 249 def ref_encode_labstate(state: dict) -> bytes:
 250     cc = sorted(state["cell_changes"], key=lambda c: (c[0], c[2], c[1]))
 251     b = bytearray(b"WLLABSV1")
 252     b += state["steps"].to_bytes(4, "little") + state["day"].to_bytes(2, "little")
 253     b += bytes((state["seconds"], 0)) + state["rng_state"].to_bytes(4, "little")
 254     b += len(cc).to_bytes(2, "little")
 255     for c in cc:
 256         b += bytes(c)
 257     b += (sum(b) & 0xFFFF).to_bytes(2, "little")
 258     return bytes(b)

 261 def pack4(rows: list[list[int]]) -> bytes:
 262     out = bytearray()
 263     for row in rows:
 264         for x in range(0, len(row), 2):
 265             out.append((row[x] << 4) | row[x + 1])
 266     return bytes(out)

============================== labstate.py
   1 """LABSTATE.BIN (SPEC-04 LR-57).
   2 
   3 Little-endian file: magic ``WLLABSV1``, step counter, day, clock seconds, a
   4 pad byte, the generator state, the cell-change count and the cell changes
   5 sorted by (location, y, x), then a u16 sum of all earlier bytes.
   6 """
   7 
   8 from __future__ import annotations
   9 
  10 from .errors import FormatError
  11 
  12 MAGIC = b"WLLABSV1"
  13 
  14 
  15 def encode_labstate(state: dict) -> bytes:
  16     """LR-57: cell changes written sorted by (location, y, x)."""
  17     cc = sorted(state.get("cell_changes") or [], key=lambda c: (c[0], c[2], c[1]))
  18     b = bytearray(MAGIC)
  19     b += int(state.get("steps", 0)).to_bytes(4, "little")
  20     b += int(state.get("day", 1)).to_bytes(2, "little")
  21     b += bytes((int(state.get("seconds", 0)) & 0xFF, 0))
  22     b += int(state.get("rng_state", 0)).to_bytes(4, "little")
  23     b += len(cc).to_bytes(2, "little")
  24     for c in cc:
  25         b += bytes(int(v) & 0xFF for v in c)
  26     b += (sum(b) & 0xFFFF).to_bytes(2, "little")
  27     return bytes(b)
  28 
  29 
  30 def decode_labstate(data: bytes) -> dict:
  31     """LR-57; ``FormatError`` on a wrong magic or sum."""
  32     if len(data) < 0x16 + 2:
  33         raise FormatError("LABSTATE: %d bytes, too short" % len(data))
  34     if data[:8] != MAGIC:
  35         raise FormatError("LABSTATE: bad magic %r" % (bytes(data[:8]),))
  36     n = int.from_bytes(data[0x14:0x16], "little")
  37     end = 0x16 + 6 * n
  38     if len(data) != end + 2:
  39         raise FormatError("LABSTATE: %d bytes, expected %d" % (len(data), end + 2))
  40     if sum(data[:end]) & 0xFFFF != int.from_bytes(data[end:end + 2], "little"):
  41         raise FormatError("LABSTATE: bad checksum")
  42     cc = [list(data[0x16 + 6 * i:0x16 + 6 * i + 6]) for i in range(n)]
  43     return {"steps": int.from_bytes(data[8:12], "little"),
  44             "day": int.from_bytes(data[12:14], "little"),
  45             "seconds": data[14], "pad": data[15],
  46             "rng_state": int.from_bytes(data[16:20], "little"),
  47             "cell_changes": cc}

  79 def encode_record(cls: int, r: dict) -> bytes:
  80     nc, na = r.get("newClass", NO_CHANGE), r.get("newAction", 0)
  81     if cls == 1:  # PRINT
  82         msgs = r["messages"]
  83         b = bytes(m | (0x80 if i == len(msgs) - 1 else 0) for i, m in enumerate(msgs))
  84         return b + _tail(nc, na)
  85     if cls == 3:  # FIXED_ENCOUNTER
  86         b = bytes((r["visibleDistance"], r["hitDistance"], r["message"]))
  87         for mob, rnd, grp in r["groups"]:
  88             b += bytes((mob, (0x80 if rnd else 0) | grp))
  89         flags = r["properName"] | (r["friendly"] << 1) | (r["immobile"] << 2) | (r["npc"] << 4)
  90         return b + bytes((flags,)) + _tail(nc, na)
  91     if cls == 4:  # MASK (lab door)
  92         return bytes((r["message"], (0x80 if r["impassable"] else 0) | r["tile"])) + _tail(nc, na)
  93     if cls == 5:  # LOOT (wllib layout)
  94         b = bytes((nc if nc != NO_CHANGE else 0, na))
  95         for iid, fixed_, amt in r["items"]:
  96             b += bytes(((0x80 if fixed_ else 0) | iid, amt))
  97         if r.get("cash") is not None:
  98             fixed_, amt = r["cash"]
  99             b += bytes((0xDE if fixed_ else 0x5E,)) + _u16(amt)
 100         return b + b"\xFF"
 101     if cls == 6:  # SHOP (lab)
 102         return bytes((r["shop"], r["message"])) + _tail(nc, na)
 103     if cls == 10:  # TRANSITION
 104         flags = (0x80 if r["relative"] else 0) | (0x40 if r["confirm"] else 0) | r["message"]
 105         return bytes((flags, r["x"] & 0xFF, r["y"] & 0xFF, r["location"])) + _tail(nc, na)
 106     if cls == 11:  # BLOCKED
 107         return bytes((r["message"],)) + _tail(nc, na)
 108     raise FormatError("scenario: lab rules define no record for class %d" % cls)

  14 def build_map(m: dict, disk: int = 0):
  15     """Return ``(block, tile_map_offset, facts)`` for one map dict."""
  16     S = m["size"]
  17     cls, act, img = m["class"], m["action"], m["image"]
  18     body = bytearray()
  19     for y in range(S):
  20         for x in range(0, S, 2):
  21             body.append((cls[y][x] << 4) | cls[y][x + 1])  # high nibble = even x
  22     for y in range(S):
  23         body += bytes(act[y])
  24     cd_at = len(body)
  25     body += bytes(42)
  26     body += encode_mapinfo(S, m["info"])
  27 
  28     offsets = [0] * 16
  29 
  30     def emit_table(c: int) -> None:
  31         recs = m["actions"].get(c)
  32         if not recs:
  33             return
  34         start = len(body)
  35         offsets[c] = start
  36         encs = [None] + [encode_record(c, r) for r in recs]  # slot 0 is null
  37         pos = start + 2 * len(encs)
  38         ptrs = []
  39         for e in encs:
  40             if e is None:
  41                 ptrs.append(0)
  42             else:
  43                 ptrs.append(pos)
  44                 pos += len(e)
  45         for p in ptrs:
  46             body.extend(_u16(p))
  47         for e in encs:
  48             if e is not None:
  49                 body.extend(e)
  50 
  51     for c in TABLE_ORDER_A:
  52         emit_table(c)
  53     npc_off = None
  54     if m["npcs"]:
  55         npc_off = len(body)
  56         n = len(m["npcs"])
  57         body.extend(_u16(0))
  58         first = npc_off + 2 + 2 * n
  59         for i in range(n):
  60             body.extend(_u16(first + i * 0x100))
  61         for ch in m["npcs"]:
  62             body.extend(containers.encode_character(ch))
  63     for c in TABLE_ORDER_B:
  64         emit_table(c)
  65     specials_off = 0  # lab rule: no special-action table
  66     names_off = len(body)
  67     for mob in m["mobs"]:
  68         body.extend(mob["name"] + b"\x00")
  69     data_off = len(body)
  70     for mob in m["mobs"]:
  71         body.extend(encode_mob(mob))
  72     strings_off = len(body)
  73     if npc_off is None:
  74         npc_off = strings_off  # lab rule: no NPCs -> npcOffset = stringsOffset
  75     body.extend(strings.encode_table(m["dictionary"], m["strings"]))
  76 
  77     cd = _u16(strings_off) + _u16(names_off) + _u16(data_off)
  78     for o in offsets:
  79         cd += _u16(o)
  80     cd += _u16(specials_off) + _u16(npc_off)
  81     body[cd_at:cd_at + 42] = cd
  82 
  83     prefix = bytes(body[:strings_off])
  84     if not codecs.rotating_xor_prefix_is_unambiguous(prefix):
  85         raise FormatError("scenario: map %r has an ambiguous encrypted prefix"
  86                           % m.get("name", "?"))
  87     enc = codecs.rotating_xor_encode(prefix)
  88     data_section = enc + bytes(body[strings_off:]) + b"\x00"  # final byte: unknown, lab 0
  89     tile_map = bytes(v for row in img for v in row)
  90     tsec = _u32(S * S) + _u32(0) + huffman.encode(tile_map)

 220 def _encode_labtables(t: dict) -> bytes:
 221     out = bytearray(containers.LABTABLES_SIZE)
 222     out[0:8] = containers.LABTABLES_MAGIC
 223     for k, (off, fmt, n) in containers.LT.items():
 224         vals = t[k] if isinstance(t[k], list) else [t[k]]
 225         if len(vals) != n:
 226             raise FormatError("LABTABLES: %s has %d values, expected %d" % (k, len(vals), n))
 227         w = containers._WIDTH[fmt]
 228         for i, v in enumerate(vals):
 229             out[off + i * w:off + (i + 1) * w] = (v & ((1 << (8 * w)) - 1)).to_bytes(w, "little")
 230     return bytes(out)

 262 def _labstr_file(tables: list) -> bytes:
 263     n = len(tables)
 264     head = strings.LABSTR_MAGIC + _u16(n) + _u16(0)
 265     pos = len(head) + 8 * n
 266     offs, sizes = b"", b""
 267     for t in tables:
 268         offs += _u32(pos)
 269         sizes += _u32(len(t))
 270         pos += len(t)

  24 def _planes_encode(pix: list, bytes_per_row: int) -> bytes:
  25     """Four 1-bit planes (blue, green, red, intensity), MSB = leftmost pixel."""
  26     out = bytearray()
  27     for bit in range(4):
  28         for row in pix:
  29             for bx in range(bytes_per_row):
  30                 v = 0
  31                 for i in range(8):
  32                     v |= ((row[bx * 8 + i] >> bit) & 1) << (7 - i)
  33                 out.append(v)
  34     return bytes(out)
  35 
  36 
  37 def _mask_encode(mask: list) -> bytes:
  38     out = bytearray()
  39     for row in mask:
  40         for bx in range(2):
  41             v = 0
  42             for i in range(8):
  43                 v |= (row[bx * 8 + i] & 1) << (7 - i)
  44             out.append(v)
  45     return bytes(out)
  46 
  47 
  48 def _cursor_encode(pix: list, vis: list) -> bytes:
  49     out = bytearray()
  50     for bit in range(4):
  51         for y in range(16):
  52             def byte(src, x0, fn):
  53                 v = 0
  54                 for i in range(8):
  55                     v |= (fn(src[y][x0 + i]) & 1) << (7 - i)
  56                 return v
  57             out += bytes((byte(vis, 8, int), byte(vis, 0, int),
  58                           byte(pix, 8, lambda p: p >> bit), byte(pix, 0, lambda p: p >> bit)))
  59     return bytes(out)
  60

  62 def _portrait_anim_encode(scripts: list, updates: list) -> bytes:
  63     sb = bytearray()
  64     for sc in scripts:
  65         for d, u in sc:
  66             sb += bytes((d, u))
  67         sb.append(0xFF)
  68     ub = bytearray()
  69     for upd in updates:
  70         for off, data in upd:
  71             ub += _u16(((len(data) - 1) << 12) | off) + data
  72         ub += _u16(0xFFFF)
  73     return _u16(len(sb)) + bytes(sb) + _u16(len(ub)) + bytes(ub)
  74 
  75 
  76 def _ending_anim_encode(updates: list) -> bytes:
  77     st = bytearray()
  78     for delay, patches in updates:
  79         st += _u16(delay)
  80         for off, data in patches:
  81             st += _u16(off) + data
  82         st += _u16(0xFFFF)
  83     st += _u16(0xFFFF)
  84     return _u16(len(st)) + bytes(st) + _u16(0)

  87 def _placeholder_tile(t: int) -> list:
  88     """Tile ``t``: fill ``t mod 16``, one-pixel border ``(t + 8) mod 16``."""
  89     pix = [[t % 16 for _ in range(16)] for _ in range(16)]
  90     border = (t + 8) % 16
  91     for x in range(16):
  92         pix[0][x] = border
  93         pix[15][x] = border
  94     for y in range(16):
  95         pix[y][0] = border
  96         pix[y][15] = border
  97     return pix
  98

 100 def _placeholder_portrait(p: int) -> list:
 101     """Portrait ``p``: fill ``4 + p``, centred 48 x 42 rectangle of colour 15."""
 102     pix = [[4 + p for _ in range(96)] for _ in range(84)]
 103     for y in range(21, 63):
 104         for x in range(24, 72):
 105             pix[y][x] = 15
 106     return pix

 109 def _build_allhtds(scenario_dir: str) -> bytes:

 116 def _build_allpics(scenario_dir: str):
 117     portraits = _load_json(os.path.join(scenario_dir, "portraits.json"))["portraits"]
 118     out = bytearray()
 119     offsets = []
 120     for p in sorted(portraits, key=lambda k: k["raw_id"]):
 121         base = graphics.pack4(_placeholder_portrait(p["raw_id"] - 1))
 122         upd = [(2000, b"\xff" * 8)]
 123         anim = _portrait_anim_encode([[(20, 0), (20, 0)]], [upd])
 124         offsets.append(len(out))
 125         out += huffman.encode_msq(codecs.vertical_xor_encode(base, PORTRAIT_STRIDE), 0)
 126         out += huffman.encode_msq(anim, 0)

 130 def _build_sprites():
 131     sprites, masks = bytearray(), bytearray()
 132     for n in range(10):
 133         pix = [[(n + x // 4 + y // 4) % 16 for x in range(16)] for y in range(16)]
 134         msk = [[1 if (x - 7.5) ** 2 + (y - 7.5) ** 2 > 56 else 0
 135                 for x in range(16)] for y in range(16)]
 136         sprites += _planes_encode(pix, 2)
 137         masks += _mask_encode(msk)
 138     return bytes(sprites), bytes(masks)
 139

 141 def _build_font() -> bytes:
 142     out = bytearray()
 143     for g in range(32):
 144         pix = [[((g >> (x % 5)) & 1) * ((g + y) % 15 + 1) for x in range(8)]
 145                for y in range(8)]
 146         out += _planes_encode(pix, 1)
 147     return bytes(out)
 148 
 149 
 150 def _build_cursors() -> bytes:
 151     out = bytearray()
 152     for k in range(2):
 153         pix = [[(x + y + k) % 16 for x in range(16)] for y in range(16)]
 154         vis = [[1 if x <= y + k * 3 else 0 for x in range(16)] for y in range(16)]
 155         out += _cursor_encode(pix, vis)
 156     return bytes(out)
 157 
 158 
 159 def _build_title() -> bytes:
 160     pix = [[((x // 18) ^ (y // 16)) % 16 for x in range(288)] for y in range(128)]
 161     return codecs.vertical_xor_encode(graphics.pack4(pix), TITLE_STRIDE)
 162 
 163 
 164 def _build_ending() -> bytes:
 165     base = graphics.pack4([[((x + y) // 24) % 16 for x in range(288)] for y in range(128)])
 166     updates = [(5, [(41, bytes.fromhex("11223344")), (40 * 100 + 35, bytes.fromhex("FFEEDDCC"))]),
 167                (9, [(0, bytes.fromhex("0F0F0F0F"))])]
 168     anim = _ending_anim_encode(updates)
 169     return (huffman.encode_msq(codecs.vertical_xor_encode(base, TITLE_STRIDE), 0)
 170             + huffman.encode_msq(anim, 0, ENDING_SIGNATURE))

 187 def _call(mod, names, *args, **kwargs):
 188     for n in names:
 189         fn = getattr(mod, n, None)
 190         if callable(fn):
 191             return fn(*args, **kwargs)
 192     raise FormatError("scenario: %s has none of %r" % (mod.__name__, names))

 134 def _compile(scenario_dir: str, out_dir: str) -> None:
 135     """Compile the scenario in ``scenario_dir`` into ``out_dir`` (SPEC-05 13)."""
 136     scen = _load_json(os.path.join(scenario_dir, "scenario.json"))
 137     items = _load_json(os.path.join(scenario_dir, "items.json"))["items"]
 138     catalog = {it["id"]: it for it in items}
 139     ranks = scen["labstr_tables"]["3"]["strings"]
 140     os.makedirs(out_dir, exist_ok=True)
 141 
 142     game = bytearray()
 143     map_facts = []
 144     for loc in scen["locations"]:
 145         m = _load_json(os.path.join(scenario_dir, loc["file"]))
 146         mm = _map_from_json(m, loc["size"], catalog, ranks)
 147         mm["name"] = loc["name"]
 148         block, tmo, facts = build_map(mm, 0)
 149         map_facts.append((len(game), tmo, facts, loc))
 150         game += block
 151 
 152     sg_off = len(game)
 153     game += _build_savegame(scenario_dir, scen, catalog, ranks)
 154 
 155     shops = _load_json(os.path.join(scenario_dir, "shops.json"))["shops"]
 156     for sh in shops:
 157         game += containers.encode_shop_list(_shop_items(sh, catalog), 0)
 158     _write(out_dir, "GAME1", bytes(game))
 159 
 160     allhtds = _build_allhtds(scenario_dir)
 161     allpics, portrait_offsets = _build_allpics(scenario_dir)
 162     _write(out_dir, "ALLHTDS1", allhtds)
 163     _write(out_dir, "ALLPICS1", allpics)
 164 
 165     sprites, masks = _build_sprites()
 166     _write(out_dir, "IC0_9.WLF", sprites)
 167     _write(out_dir, "MASKS.WLF", masks)
 168     _write(out_dir, "COLORF.FNT", _build_font())
 169     _write(out_dir, "CURS", _build_cursors())
 170     _write(out_dir, "TITLE.PIC", _build_title())
 171     _write(out_dir, "END.CPA", _build_ending())
 172

 174            _build_labtables(map_facts, sg_off, len(allhtds), portrait_offsets))

   1 """Map blocks (SPEC-03).
   2 
   3 A map block lives inside GAME1 or GAME2: an ``msq`` header, a rotating-XOR
   4 encrypted prefix followed by plain bytes (the map-data body), then the Huffman
   5 coded tile map.  ``decode_map`` reads one block; ``load_map`` finds one through
   6 LABTABLES.BIN (SPEC-02 section 3.4).
   7 
   8 The rest of the module implements the lab rules of SPEC-03 section 9 on the
   9 decoded dict: ``record_at``, ``is_passable``, ``apply_tail``,
  10 ``resolve_transition`` and ``image_at``.
  11 """

  33 def _r16(b: bytes, o: int) -> int:
  34     if o < 0 or o + 2 > len(b):
  35         raise FormatError("map: u16 at offset %d runs past the body" % o)
  36     return int.from_bytes(b[o:o + 2], "little")

  39 def _r32(b: bytes, o: int) -> int:
  40     if o < 0 or o + 4 > len(b):
  41         raise FormatError("map: u32 at offset %d runs past the input" % o)
  42     return int.from_bytes(b[o:o + 4], "little")

  53 def _read_tail(b: bytes, o: int, r: dict) -> int:
  54     """Common tail (8.3): u8 newActionClass, u8 newAction when < 253."""
  55     if o >= len(b):
  56         raise FormatError("map: record tail runs past the body")
  57     nc = b[o]
  58     o += 1
  59     if nc < 253:
  60         if o >= len(b):
  61             raise FormatError("map: record tail runs past the body")
  62         r["newClass"], r["newAction"] = nc, b[o]
  63         o += 1
  64     else:
  65         r["newClass"], r["newAction"] = nc, 0
  66     return o

  69 def decode_record(cls: int, b: bytes, o: int) -> dict:
  70     """Decode one action record of class ``cls`` at body offset ``o``."""
  71     if o < 0 or o >= len(b):
  72         raise FormatError("map: record offset %d outside the body" % o)
  73     r: dict = {}
  74     if cls == 1:  # PRINT (8.4)
  75         msgs = []
  76         while True:
  77             if o >= len(b):
  78                 raise FormatError("map: PRINT record runs past the body")
  79             m = b[o]
  80             o += 1
  81             msgs.append(m & 0x7F)
  82             if m & 0x80:
  83                 break
  84         r["messages"] = msgs
  85         _read_tail(b, o, r)
  86     elif cls == 3 or cls == 15:  # FIXED_ENCOUNTER / RANDOM_ENCOUNTER (8.5)
  87         if o + 10 > len(b):
  88             raise FormatError("map: encounter record runs past the body")
  89         r["visibleDistance"], r["hitDistance"], r["message"] = b[o], b[o + 1], b[o + 2]
  90         o += 3
  91         groups = []
  92         for _ in range(3):
  93             groups.append([b[o], bool(b[o + 1] & 0x80), b[o + 1] & 0x7F])
  94             o += 2
  95         r["groups"] = groups
  96         f = b[o]
  97         o += 1
  98         r.update({"properName": f & 1, "friendly": (f >> 1) & 1,
  99                   "immobile": (f >> 2) & 1, "npc": f >> 4})
 100         _read_tail(b, o, r)
 101     elif cls == 4:  # MASK (8.7)
 102         if o + 2 > len(b):
 103             raise FormatError("map: MASK record runs past the body")
 104         r["message"], r["impassable"], r["tile"] = b[o], bool(b[o + 1] & 0x80), b[o + 1] & 0x7F
 105         _read_tail(b, o + 2, r)
 106     elif cls == 5:  # LOOT (8.8, wllib layout)
 107         if o + 2 > len(b):
 108             raise FormatError("map: LOOT record runs past the body")
 109         r["newClass"], r["newAction"] = b[o], b[o + 1]
 110         o += 2
 111         r["items"], r["cash"] = [], None
 112         while True:
 113             if o >= len(b):
 114                 raise FormatError("map: LOOT record has no 0xFF terminator")
 115             v = b[o]
 116             if v == 0xFF:
 117                 break
 118             if v in (0xDE, 0x5E):
 119                 if o + 3 > len(b):
 120                     raise FormatError("map: LOOT cash entry runs past the body")
 121                 r["cash"] = [v == 0xDE, _r16(b, o + 1)]
 122                 o += 3
 123             else:
 124                 if o + 2 > len(b):
 125                     raise FormatError("map: LOOT item entry runs past the body")
 126                 r["items"].append([v & 0x7F, bool(v & 0x80), b[o + 1]])
 127                 o += 2
 128     elif cls == 6:  # SHOP (lab layout, 9.4)
 129         if o + 2 > len(b):
 130             raise FormatError("map: SHOP record runs past the body")
 131         r["shop"], r["message"] = b[o], b[o + 1]
 132         _read_tail(b, o + 2, r)
 133     elif cls == 10:  # TRANSITION (8.9)
 134         if o + 4 > len(b):
 135             raise FormatError("map: TRANSITION record runs past the body")
 136         f = b[o]
 137         r.update({"relative": bool(f & 0x80), "confirm": bool(f & 0x40),
 138                   "message": f & 0x3F, "x": _i8(b[o + 1]), "y": _i8(b[o + 2]),
 139                   "location": b[o + 3]})
 140         _read_tail(b, o + 4, r)
 141     elif cls == 11:  # BLOCKED (8.10)
 142         if o + 1 > len(b):
 143             raise FormatError("map: BLOCKED record runs past the body")
 144         r["message"] = b[o]
 145         _read_tail(b, o + 1, r)
 146     else:
 147         raise FormatError("map: class %d has no record layout" % cls)
 148     return r

 155 def _decode_pointer_table(body: bytes, start: int, cls: int) -> list:
 156     """Pointer list at ``start`` (SPEC-03 6.2), records decoded per class.
 157 
 158     Pointers are read until the read position reaches the smallest non-zero
 159     pointer seen so far (the wastelib rule); lab data keeps pointers ascending
 160     so the wllib rule agrees.
 161     """
 162     ptrs = []
 163     pos = start
 164     end_ptr = None
 165     while end_ptr is None or pos < end_ptr:
 166         p = _r16(body, pos)
 167         ptrs.append(p)
 168         pos += 2
 169         if p and (end_ptr is None or p < end_ptr):
 170             end_ptr = p
 171     out = []
 172     for p in ptrs:
 173         if p == 0:
 174             out.append(None)
 175         else:
 176             if p >= len(body):
 177                 raise FormatError("map: action pointer %d outside the body" % p)
 178             out.append(decode_record(cls, body, p))
 179     return out

 182 def decode_map(game: bytes, offset: int, map_size: int, tile_map_offset: int) -> dict:
 183     """Decode the map block at ``offset`` (SPEC-03 sections 3 to 7)."""
 184     if offset < 0 or offset + 4 > len(game):
 185         raise FormatError("map: block header runs past the end of the file")
 186     if game[offset:offset + 3] != b"msq" or game[offset + 3] not in (0x30, 0x31):
 187         raise FormatError("map: bad block header %s"
 188                           % bytes(game[offset:offset + 4]).hex(" ").upper())
 189     if map_size not in (32, 64):
 190         raise FormatError("map: unsupported map size %d" % map_size)
 191     if tile_map_offset < 7:
 192         raise FormatError("map: tile-map offset %d is too small" % tile_map_offset)
 193 
 194     body_size = tile_map_offset - 7
 195     body, plen = codecs.rotating_xor_decode_prefix(game, body_size, offset + 4)
 196 
 197     t = offset + tile_map_offset
 198     tsize = _r32(game, t)
 199     if tsize != map_size * map_size:
 200         raise FormatError("map: tile-map size %d != %d" % (tsize, map_size * map_size))
 201     tiles, _end = huffman.decode(game, tsize, t + 8)
 202 
 203     n = map_size * map_size
 204     cm = body[:n // 2]
 205     am = body[n // 2:n // 2 + n]
 206     if len(cm) < n // 2 or len(am) < n:
 207         raise FormatError("map: body is too short for a %d x %d map" % (map_size, map_size))
 208     cdo = n // 2 + n
 209     if cdo + CD_SIZE + MAPINFO_SIZE > len(body):
 210         raise FormatError("map: central directory or map info runs past the body")
 211     cd = body[cdo:cdo + CD_SIZE]
 212     sd = {
 213         "stringsOffset": _r16(cd, 0),
 214         "mobNamesOffset": _r16(cd, 2),
 215         "mobDataOffset": _r16(cd, 4),
 216         "actionClassOffsets": [_r16(cd, 6 + 2 * i) for i in range(16)],
 217         "specialsOffset": _r16(cd, 0x26),
 218         "npcOffset": _r16(cd, 0x28),
 219     }
 220     for name in ("stringsOffset", "mobNamesOffset", "mobDataOffset", "specialsOffset", "npcOffset"):
 221         v = sd[name]
 222         if v and v >= len(body):
 223             raise FormatError("map: central-directory %s %d outside the body" % (name, v))
 224     for c, v in enumerate(sd["actionClassOffsets"]):
 225         if v and v >= len(body):
 226             raise FormatError("map: action-class offset %d (class %d) outside the body" % (v, c))
 227 
 228     info = _decode_mapinfo(body[cdo + CD_SIZE:cdo + CD_SIZE + MAPINFO_SIZE])
 229     if info["mapSize"] != map_size:
 230         raise FormatError("map: map info size %d != %d" % (info["mapSize"], map_size))
 231 
 232     # L-EN2: in lab data the encrypted prefix ends exactly at stringsOffset.
 233     if plen != sd["stringsOffset"]:
 234         raise FormatError("map: encrypted prefix ends at %d, stringsOffset is %d (L-EN2)"
 235                           % (plen, sd["stringsOffset"]))
 236 
 237     cls = [[(cm[(y * map_size + x) >> 1] >> 4) if x % 2 == 0
 238             else (cm[(y * map_size + x) >> 1] & 0x0F)
 239             for x in range(map_size)] for y in range(map_size)]
 240     act = [list(am[y * map_size:(y + 1) * map_size]) for y in range(map_size)]
 241     img = [list(tiles[y * map_size:(y + 1) * map_size]) for y in range(map_size)]
 242 
 243     actions = {}
 244     for c in range(16):
 245         o = sd["actionClassOffsets"][c]
 246         if o == 0:
 247             continue
 248         actions[str(c)] = _decode_pointer_table(body, o, c)
 249 
 250     npcs = []
 251     o = sd["npcOffset"]
 252     if o and _r16(body, o) == 0:
 253         pos, lo, ptrs = o + 2, None, []
 254         while lo is None or pos < lo:
 255             p = _r16(body, pos)
 256             ptrs.append(p)
 257             pos += 2
 258             lo = p if lo is None or p < lo else lo
 259         for p in ptrs:
 260             if p + 0x100 > len(body):
 261                 raise FormatError("map: NPC pointer %d outside the body" % p)
 262             npcs.append(containers.decode_character(body[p:p + 0x100]))
 263 
 264     nm = (sd["stringsOffset"] - sd["mobDataOffset"]) // 8
 265     names = []
 266     p = sd["mobNamesOffset"]
 267     for _ in range(nm):
 268         e = body.find(b"\x00", p)
 269         if e < 0:
 270             raise FormatError("map: monster name %d is not NUL-terminated" % len(names))
 271         names.append(body[p:e])
 272         p = e + 1
 273     mobs = []
 274     for i in range(nm):
 275         o = sd["mobDataOffset"] + 8 * i
 276         if o + 8 > len(body):
 277             raise FormatError("map: monster data runs past the body")
 278         mobs.append(_decode_mob(body[o:o + 8], names[i]))
 279 
 280     decoded = strings.decode_table(body, sd["stringsOffset"])
 281     return {
 282         "size": map_size,
 283         "bodySize": body_size,
 284         "encryptedPrefixLength": plen,
 285         "centralDirectory": sd,
 286         "info": info,
 287         "cells": {"actionClass": cls, "action": act, "image": img},
 288         "actions": actions,
 289         "npcs": npcs,
 290         "mobs": mobs,
 291         "strings": [s.decode("latin-1") for s in decoded],
 292     }
 293

 295 def _decode_mapinfo(b: bytes) -> dict:
 296     if len(b) < MAPINFO_SIZE:
 297         raise FormatError("map: map-info block is shorter than 50 bytes")
 298     d = {"unknown00": b[0], "unknown01": b[1], "mapSize": b[2],
 299          "unknown03": b[3], "unknown04": b[4]}
 300     for i, k in enumerate(MAPINFO_FIELDS):
 301         d[k] = b[5 + i]
 302     d["timePerStep"] = _r16(b, 0x0A)
 303     d["healRate"] = b[0x0C]
 304     d["combatStringIds"] = list(b[0x0D:0x32])
 305     return d

 311             "armorClass": b[4] & 15, "fixedDamage": b[5] >> 4,
 312             "weaponType": b[5] & 15, "type": b[6], "portrait": b[7]}

 319 def load_map(data_dir: str, location: int) -> dict:
 320     """Load the map for ``location`` through LABTABLES.BIN and GAME1/GAME2."""
 321     import os
 322 
 323     tables = containers.read_tables(os.path.join(data_dir, "LABTABLES.BIN"))
 324     loc, disk, index = containers.resolve_location(tables, location)
 325     if disk == 2:
 326         name, off_index = "GAME1", index
 327     elif disk == 1:
 328         name, off_index = "GAME2", 20 + index
 329     else:
 330         raise FormatError("map: location %d has an invalid disk code %d" % (location, disk))
 331     with open(os.path.join(data_dir, name), "rb") as fh:
 332         game = fh.read()
 333     offsets = tables["mapOffsets"]
 334     if off_index >= len(offsets):
 335         raise FormatError("map: map index %d is outside mapOffsets" % off_index)
 336     offset = offsets[off_index]
 337     sizes = tables["mapSizes"]
 338     if loc >= len(sizes):
 339         raise FormatError("map: location %d is outside mapSizes" % loc)
 340     size = sizes[loc]
 341     tmos = tables["tileMapOffsets"]
 342     if loc >= len(tmos):
 343         raise FormatError("map: location %d is outside tileMapOffsets" % loc)
 344     return decode_map(game, offset, size, tmos[loc])

 351 def record_at(map: dict, x: int, y: int):
 352     """The record for cell ``(x, y)``, or ``None`` (L-AC2, L-AC3, L-AC5)."""
 353     size = map["size"]
 354     if not (0 <= x < size and 0 <= y < size):
 355         raise FormatError("map: cell (%d, %d) outside the %d x %d map" % (x, y, size, size))
 356     cls = map["cells"]["actionClass"][y][x]
 357     a = map["cells"]["action"][y][x]
 358     if a == 0:
 359         if cls in NEEDS_RECORD:
 360             raise FormatError("map: class %d cell (%d, %d) has action byte 0 (L-AC5)"
 361                               % (cls, x, y))
 362         return None
 363     table = map["actions"].get(str(cls))
 364     if table is None:
 365         return None
 366     if a >= len(table):
 367         raise FormatError("map: class %d cell (%d, %d) has action byte %d beyond the "
 368                           "pointer list (%d entries) (L-AC5)" % (cls, x, y, a, len(table)))
 369     rec = table[a]
 370     if rec is None and cls in NEEDS_RECORD:
 371         raise FormatError("map: class %d cell (%d, %d) points at a null record (L-AC5)"
 372                           % (cls, x, y))
 373     return rec

 376 def is_passable(map: dict, x: int, y: int) -> bool:
 377     """Whether the party may step onto ``(x, y)`` (9.1, 9.6, L-MV1, L-MV2)."""
 378     size = map["size"]
 379     if not (0 <= x < size and 0 <= y < size):
 380         return False
 381     cls = map["cells"]["actionClass"][y][x]
 382     if cls == 0:
 383         return True
 384     if cls == 11:
 385         return False
 386     if cls == 4:
 387         rec = record_at(map, x, y)
 388         return True if rec is None else not rec["impassable"]
 389     if cls in (1, 3, 5, 6, 10):
 390         return True
 391     return True  # reserved classes are walkable (L-AC1)

 394 def apply_tail(record):
 395     """``(newClass, newAction)`` for a record tail, or ``None`` (L-TL1)."""
 396     nc = record.get("newClass", 255)
 397     if nc >= 253:
 398         return None
 399     if nc >= 16:
 400         return None
 401     return [nc, record.get("newAction", 0)]

 404 def resolve_transition(tables: dict, record: dict, x: int, y: int, previous):
 405     """Resolve a TRANSITION record to ``(location, x, y)`` (L-TR2, L-TR3, L-TR5)."""
 406     t = record["location"]
 407     if t == 255:
 408         return (previous[0], previous[1], previous[2])
 409     if 128 <= t <= 252:
 410         shared = tables["sharedLocations"]
 411         if t - 128 >= len(shared):
 412             raise FormatError("map: shared location %d has no entry" % t)
 413         t = shared[t - 128]
 414     loc, _disk, _index = containers.resolve_location(tables, t)
 415     if record.get("relative"):
 416         tx = x + record["x"]
 417         ty = y + record["y"]
 418     else:
 419         tx = record["x"] & 0xFF
 420         ty = record["y"] & 0xFF
 421     size = tables["mapSizes"][loc]
 422     if not (0 <= tx < size and 0 <= ty < size):
 423         raise FormatError("map: transition target (%d, %d) outside the %d x %d map (L-TR5)"
 424                           % (tx, ty, size, size))
 425     return [loc, tx, ty]

 428 def image_at(map: dict, x: int, y: int) -> int:
 429     """The cell image, or the border image outside the map (L-MI5)."""
 430     size = map["size"]
 431     if not (0 <= x < size and 0 <= y < size):
 432         return map["info"]["borderImage"]
 433     return map["cells"]["image"][y][x]
