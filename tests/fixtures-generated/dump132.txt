 204         """Return (events, data_dir). Fails the test cleanly on any engine problem."""

 219             err = str(e)
 220         self.cache[ck] = (ev, dd, err)
 221         if err:
 222             W.fail(err)

  59 def norm(v):
  60     """Normalise an engine value for comparison: tuples become lists, integer
  61     dict keys become strings, bytearray becomes bytes. Values are untouched."""
  62     if isinstance(v, dict):
  63         return {(str(k) if isinstance(k, int) else k): norm(x) for k, x in v.items()}
  64     if isinstance(v, (list, tuple)):
  65         return [norm(x) for x in v]
  66     if isinstance(v, bytearray):
  67         return bytes(v)
  68     return v

  71 def first_diff(got, want, path: str = ""):
  72     """Return (path, got_part, want_part) of the first difference, or None."""
  73     if isinstance(want, dict):
  74         if not isinstance(got, dict):
  75             return (path or "<root>", got, "a dict")
  76         for k in want:
  77             if k not in got:
  78                 return ("%s.%s" % (path, k) if path else k, "<missing key>", want[k])
  79             d = first_diff(got[k], want[k], "%s.%s" % (path, k) if path else str(k))
  80             if d:
  81                 return d
  82         extra = [k for k in got if k not in want]
  83         if extra:
  84             return ("%s.%s" % (path, extra[0]) if path else str(extra[0]), got[extra[0]], "<no such key>")
  85         return None
  86     if isinstance(want, list):
  87         if not isinstance(got, list):
  88             return (path or "<root>", got, "a list of %d items" % len(want))
  89         for i, (g, w) in enumerate(zip(got, want)):
  90             d = first_diff(g, w, "%s[%d]" % (path, i))
  91             if d:
  92                 return d
  93         if len(got) != len(want):
  94             return ("%s (length)" % (path or "<root>"), len(got), len(want))
  95         return None
  96     return None if got == want else (path or "<root>", got, want)

  99 def check_eq(what: str, got, want) -> None:
 100     """Compare after normalising; report only the first differing element."""
 101     g = norm(got)
 102     if g == want:
 103         return
 104     d = first_diff(g, want)
 105     if d is None:  # equal under first_diff's rules but not ==, report whole case
 106         fail("%s: expected %s, got %s" % (what, short(want), short(g)))
 107     p, gv, wv = d
 108     if p in ("<root>", ""):
 109         fail("%s: expected %s, got %s" % (what, short(wv), short(gv)))
 110     fail("%s: first difference at %s: expected %s, got %s" % (what, p, short(wv), short(gv)))

 113 def check_bytes(what: str, got, want: bytes) -> None:
 114     if not isinstance(got, (bytes, bytearray)):
 115         fail("%s: expected bytes, got %s" % (what, type(got).__name__))
 116     got = bytes(got)
 117     if got == want:
 118         return
 119     n = min(len(got), len(want))
 120     i = next((k for k in range(n) if got[k] != want[k]), n)
 121     if i == n:
 122         fail("%s: expected %d bytes, got %d (common prefix identical)" % (what, len(want), len(got)))
 123     fail("%s: first differing byte at offset %d: expected %02X, got %02X (lengths: expected %d, got %d)"
 124          % (what, i, want[i], got[i], len(want), len(got)))

 224 def error_class(name: str):
 225     cls = api("errors." + name)
 226     if not (isinstance(cls, type) and issubclass(cls, Exception)):
 227         fail("wlengine.errors.%s must be an exception class" % name)
 228     if name in ("FormatError", "ChecksumError") and not issubclass(cls, ValueError):
 229         fail("wlengine.errors.%s must subclass ValueError (SPEC-07 section 3)" % name)
 230     if name == "ChecksumError" and not issubclass(cls, api("errors.FormatError")):
 231         fail("wlengine.errors.ChecksumError must subclass FormatError (SPEC-07 section 3)")
 232     if cls in (Exception, ValueError, BaseException) or issubclass(NotImplementedError, cls):
 233         fail("wlengine.errors.%s must be a dedicated class (SPEC-07 section 3)" % name)
 234     return cls

 248     """Call wlengine.<dotted>(*args, **kw); any exception becomes a clean failure."""

 319 class RefRng:
 320     """SPEC-04 LR-1 and LR-2, restated for computing expected values."""
 321 
 322     def __init__(self, seed: int) -> None:
 323         self.state = seed & 0xFFFFFFFF or 0x6D2B79F5
 324 
 325     def next(self) -> int:
 326         x = self.state
 327         x ^= (x << 13) & 0xFFFFFFFF
 328         x ^= x >> 17
 329         x ^= (x << 5) & 0xFFFFFFFF
 330         self.state = x
 331         return x
 332 
 333     def roll(self, n: int) -> int:
 334         return 1 + self.next() % n
 335 
 336     def rnd(self, n: int) -> int:
 337         if n == 0:
 338             return 0
 339         if n == 1:
 340             return 1
 341         return self.roll(n)

 431 def ref_pay(moneys: list[int], first_slot: int, price: int):
 432     if sum(moneys) < price:
 433         return None
 434     out = list(moneys)
 435     order = [first_slot - 1] + [i for i in range(len(moneys)) if i != first_slot - 1]
 436     left = price
 437     for i in order:
 438         take = min(out[i], left)
 439         out[i] -= take
 440         left -= take
 441         if left == 0:
 442             break
 443     return out

 446 def ref_advance_clock(clock: dict, minutes: int, seconds: int):
 447     day, minute, second = clock["day"], clock["minute"], clock["second"]
 448     total_s = second + seconds
 449     second = total_s % 60
 450     add_min = minutes + total_s // 60
 451     hours = []
 452     for _ in range(add_min):
 453         minute += 1
 454         if minute == 1440:
 455             minute = 0
 456         if minute % 60 == 0:
 457             h = minute // 60
 458             hours.append(h)
 459             if h == 0:
 460                 day += 1
 461     return {"day": day, "minute": minute, "second": second}, hours

 464 def ref_build_groups(rng: RefRng, mobs: list[dict], record: dict) -> list[dict]:
 465     counts = []
 466     for mob_index, rnd_flag, size in record["groups"]:
 467         if mob_index == 0:
 468             continue
 469         mob = mobs[mob_index - 1]
 470         s = size if size else mob["maxGroup"]
 471         counts.append((mob_index, rng.rnd(s) if rnd_flag else s))
 472     out = []
 473     for mob_index, count in counts:
 474         if count <= 0:
 475             continue
 476         mob = mobs[mob_index - 1]
 477         hp = [ref_monster_hp(rng, mob["sturdiness"]) for _ in range(count)]
 478         band = 3 if mob["weaponType"] >= 3 else 2
 479         out.append({"monster": mob_index, "count": count, "band": band, "hp": hp})
 480     return out

 492 def ref_labstate(blob: bytes) -> dict:
 493     """Independent LABSTATE.BIN reader (SPEC-04 LR-57)."""
 494     if blob[:8] != b"WLLABSV1":
 495         raise ValueError("bad magic")
 496     n = int.from_bytes(blob[0x14:0x16], "little")
 497     end = 0x16 + 6 * n
 498     if len(blob) != end + 2:
 499         raise ValueError("bad length")
 500     if sum(blob[:end]) & 0xFFFF != int.from_bytes(blob[end:end + 2], "little"):
 501         raise ValueError("bad sum")
 502     cc = [list(blob[0x16 + 6 * i:0x16 + 6 * i + 6]) for i in range(n)]
 503     return {"steps": int.from_bytes(blob[8:12], "little"), "day": int.from_bytes(blob[12:14], "little"),
 504             "seconds": blob[14], "pad": blob[15], "rng_state": int.from_bytes(blob[16:20], "little"),
 505             "cell_changes": cc}

 508 def ref_encode_labstate(state: dict) -> bytes:
 509     cc = sorted(state["cell_changes"], key=lambda c: (c[0], c[2], c[1]))
 510     b = bytearray(b"WLLABSV1")
 511     b += state["steps"].to_bytes(4, "little") + state["day"].to_bytes(2, "little")
 512     b += bytes((state["seconds"], 0)) + state["rng_state"].to_bytes(4, "little")
 513     b += len(cc).to_bytes(2, "little")
 514     for c in cc:
 515         b += bytes(c)
 516     b += (sum(b) & 0xFFFF).to_bytes(2, "little")
 517     return bytes(b)

 520 def pack4(rows: list[list[int]]) -> bytes:
 521     out = bytearray()
 522     for row in rows:
 523         for x in range(0, len(row), 2):
 524             out.append((row[x] << 4) | row[x + 1])
 525     return bytes(out)

 528 def unpack4(data: bytes, width: int) -> list[list[int]]:
 529     stride = width // 2
 530     return [[v for b in data[y * stride:(y + 1) * stride] for v in (b >> 4, b & 15)]
 531             for y in range(len(data) // stride)]
