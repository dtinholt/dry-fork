   15 def encode_labstate(state: dict) -> bytes:
   16     """LR-57: cell changes written sorted by (location, y, x)."""
   17     cc = sorted(state.get("cell_changes") or [], key=lambda c: (c[0], c[2], c[1]))
   18     b = bytearray(MAGIC)
   19     b += int(state.get("steps", 0)).to_bytes(4, "little")
   20     b += int(state.get("day", 1)).to_bytes(2, "little")
   21     b += bytes((int(state.get("seconds", 0)) & 0xFF, 0))
   22     b += int(state.get("rng_state", 0)).to_bytes(4, "little")
   23     b += len(cc).to_bytes(2, "little")
   24     for c in cc:
   25         b += bytes(int(v) & 0xFF for v in c)
   26     b += (sum(b) & 0xFFFF).to_bytes(2, "little")
   27     return bytes(b)

   30 def decode_labstate(data: bytes) -> dict:
   31     """LR-57; ``FormatError`` on a wrong magic or sum."""
   32     if len(data) < 0x16 + 2:
   33         raise FormatError("LABSTATE: %d bytes, too short" % len(data))
   34     if data[:8] != MAGIC:
   35         raise FormatError("LABSTATE: bad magic %r" % (bytes(data[:8]),))
   36     n = int.from_bytes(data[0x14:0x16], "little")
   37     end = 0x16 + 6 * n

   39         raise FormatError("LABSTATE: %d bytes, expected %d" % (len(data), end + 2))

def test_save_savegame_fields(runner, walkthrough):
    ev, dd = _saved(runner, walkthrough)
    p = _payload(dd)
    got = {"order": list(p[1:8]), "x": p[8], "y": p[9], "map": p[10], "prevX": p[11], "prevY": p[12], "prevMap": p[13],
           "groups1to3": list(p[14:56]), "viewport": [p[0x78], p[0x79]], "members": [p[0x7D], p[0x80]],
           "party": p[0x7E], "location": p[0x7F], "extra": p[0x81], "clock": [p[0x83], p[0x84]], "scroll": p[0x85],
           "serial": int.from_bytes(p[0xF5:0xF9], "little")}
    want = {"order": [1, 2, 3, 4, 5, 0, 0], "x": 25, "y": 5, "map": 2, "prevX": 43, "prevY": 28, "prevMap": 0,
            "groups1to3": [0] * 42, "viewport": [25, 5], "members": [5, 5], "party": 0, "location": 2, "extra": 0,

    names = [bytes(p[0x100 * (s + 1):0x100 * (s + 1) + 14]).split(b"\x00")[0].decode("latin-1") for s in range(7)]
    check_eq("character record names (LR-55)", names, ["Hessa Brand", "Tobin Ash", "Maro Quill", "Dell Ostrow",
                                                       "Orla Penn", "", ""])
    check_eq("slots 6 and 7 all zero (LR-55)", any(p[0x600:0x800]), False)
