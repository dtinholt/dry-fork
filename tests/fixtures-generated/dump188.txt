    1 """Shared helpers for the PREREG-W1 acceptance tests (SPEC-07).
    2 
    3 This module is test infrastructure, not an oracle for any single answer:
    4 
    5 - `api()` / `call()` reach into the builder's `wlengine` package and turn every
    6   engine-side problem (missing module, missing function, NotImplementedError,
    7   any other exception) into a clean test failure with a one-line message.
    8 - `check_eq()` and friends report the first difference only, so a failure never
    9   prints more of the expected data than the single case being checked.
   10 - The `ref_*` functions and `RefRng` are independent restatements of SPEC-04
   11   formulas, used to compute expected values. They are not the engine.
   12 
   13 Frozen with the tests. The engine must not import this module.
   14 """

   50 def short(v, limit: int = 160) -> str:
   51     if isinstance(v, (bytes, bytearray)):
   52         h = bytes(v).hex(" ").upper()
   53         s = "bytes[%d] %s" % (len(v), h)
   54     else:
   55         s = repr(v)
   56     return s if len(s) <= limit else s[:limit] + "..."

   59 def norm(v):
   60     """Normalise an engine value for comparison: tuples become lists, integer
   61     dict keys become strings, bytearray becomes bytes. Values are untouched."""
   62     if isinstance(v, dict):
   63         return {(str(k) if isinstance(k, int) else k): norm(x) for k, x in v.items()}
   64     if isinstance(v, (list, tuple)):
   65         return [norm(x) for x in v]
   66     if isinstance(v, bytearray):
   67         return bytes(v)
   68     return v

   71 def first_diff(got, want, path: str = ""):
   72     """Return (path, got_part, want_part) of the first difference, or None."""
   73     if isinstance(want, dict):
   74         if not isinstance(got, dict):
   75             return (path or "<root>", got, "a dict")
   76         for k in want:
   77             if k not in got:
   78                 return ("%s.%s" % (path, k) if path else k, "<missing key>", want[k])
   79             d = first_diff(got[k], want[k], "%s.%s" % (path, k) if path else str(k))
   80             if d:
   81                 return d
   82         extra = [k for k in got if k not in want]
   83         if extra:
   84             return ("%s.%s" % (path, extra[0]) if path else str(extra[0]), got[extra[0]], "<no such key>")
   85         return None
   86     if isinstance(want, list):
   87         if not isinstance(got, list):
   88             return (path or "<root>", got, "a list of %d items" % len(want))
   89         for i, (g, w) in enumerate(zip(got, want)):
   90             d = first_diff(g, w, "%s[%d]" % (path, i))
   91             if d:
   92                 return d
   93         if len(got) != len(want):
   94             return ("%s (length)" % (path or "<root>"), len(got), len(want))
   95         return None
   96     return None if got == want else (path or "<root>", got, want)

   99 def check_eq(what: str, got, want) -> None:
  100     """Compare after normalising; report only the first differing element."""
  101     g = norm(got)
  102     if g == want:
  103         return
  104     d = first_diff(g, want)
  105     if d is None:  # equal under first_diff's rules but not ==, report whole case
  106         fail("%s: expected %s, got %s" % (what, short(want), short(g)))
  107     p, gv, wv = d
  108     if p in ("<root>", ""):
  109         fail("%s: expected %s, got %s" % (what, short(wv), short(gv)))
  110     fail("%s: first difference at %s: expected %s, got %s" % (what, p, short(wv), short(gv)))
  111

  113 def check_bytes(what: str, got, want: bytes) -> None:
  114     if not isinstance(got, (bytes, bytearray)):
  115         fail("%s: expected bytes, got %s" % (what, type(got).__name__))
  116     got = bytes(got)
  117     if got == want:
  118         return
  119     n = min(len(got), len(want))
  120     i = next((k for k in range(n) if got[k] != want[k]), n)
  121     if i == n:
  122         fail("%s: expected %d bytes, got %d (common prefix identical)" % (what, len(want), len(got)))
  123     fail("%s: first differing byte at offset %d: expected %02X, got %02X (lengths: expected %d, got %d)"
  124          % (what, i, want[i], got[i], len(want), len(got)))

  137 def _scan_sources() -> str | None:
  138     """SPEC-07 section 6: src/ must not name the oracle modules or files."""
  139     if not SRC.is_dir():
  140         return None
  141     for p in sorted(SRC.rglob("*.py")):
  142         try:
  143             text = p.read_text(encoding="utf-8", errors="replace")
  144         except OSError:
  145             continue
  146         for lineno, line in enumerate(text.splitlines(), 1):
  147             for name in ORACLE_NAMES:
  148                 if name in line:
  149                     return "%s line %d names %r" % (p.relative_to(PROJECT), lineno, name)
  150     return None

  153 def engine_ready() -> None:
  154     """Fail the test unless `wlengine` imports cleanly and src/ is clean."""
  155     if not _ENGINE["checked"]:
  156         _ENGINE["checked"] = True
  157         bad = _scan_sources()
  158         if bad:
  159             _ENGINE["error"] = ("SPEC-07 rule 6.1: src/ must not reference the oracle modules or the "
  160                                 "reference log (%s)" % bad)
  161         else:
  162             try:
  163                 _armed(lambda: importlib.import_module("wlengine"))
  164             except Exception as e:  # noqa: BLE001
  165                 _ENGINE["error"] = "cannot import wlengine from src/: %s: %s" % (type(e).__name__, e)
  166     if _ENGINE["error"]:
  167         fail(_ENGINE["error"])

  170 def _armed(fn, *a, allow_reference_log=False, **k):
  171     _STATE["armed"] += 1
  172     prev = _STATE["allow_reference_log"]
  173     _STATE["allow_reference_log"] = allow_reference_log
  174     try:
  175         return fn(*a, **k)
  176     finally:
  177         _STATE["armed"] -= 1
  178         _STATE["allow_reference_log"] = prev

  181 def audit_hook(event, args):
  182     if not _STATE["armed"] or event != "open" or not args:
  183         return
  184     p = args[0]
  185     if isinstance(p, int):
  186         return
  187     try:
  188         p = os.fspath(p)
  189         if isinstance(p, bytes):
  190             p = p.decode("utf-8", "replace")
  191         p = os.path.abspath(p)
  192     except Exception:  # noqa: BLE001
  193         return
  194     for pref in FORBIDDEN_PREFIXES:
  195         if _STATE["allow_reference_log"] and pref == str(REFERENCE_LOG):
  196             continue
  197         if p == pref.rstrip(os.sep) or p.startswith(pref):
  198             _STATE["violations"].append(p)

  201 def _module_violation() -> str | None:
  202     spec_dir = str(SPEC) + os.sep
  203     for name, mod in list(sys.modules.items()):
  204         f = getattr(mod, "__file__", None)
  205         if f and str(f).startswith(spec_dir) and not name.startswith("_wl_oracle_"):
  206             return "module %r loaded from %s" % (name, f)
  207     return None

  210 def api(dotted: str):
  211     """Resolve 'module.function' inside wlengine, failing cleanly if absent."""
  212     engine_ready()
  213     mod_name, _, attr = dotted.rpartition(".")
  214     full = "wlengine." + mod_name if mod_name else "wlengine"
  215     try:
  216         mod = _armed(importlib.import_module, full)
  217     except Exception as e:  # noqa: BLE001
  218         fail("cannot import %s (SPEC-07 API): %s: %s" % (full, type(e).__name__, e))
  219     if not hasattr(mod, attr):
  220         fail("%s has no attribute %r (SPEC-07 API)" % (full, attr))
  221     return getattr(mod, attr)

  224 def error_class(name: str):
  225     cls = api("errors." + name)
  226     if not (isinstance(cls, type) and issubclass(cls, Exception)):
  227         fail("wlengine.errors.%s must be an exception class" % name)
  228     if name in ("FormatError", "ChecksumError") and not issubclass(cls, ValueError):
  229         fail("wlengine.errors.%s must subclass ValueError (SPEC-07 section 3)" % name)
  230     if name == "ChecksumError" and not issubclass(cls, api("errors.FormatError")):
  231         fail("wlengine.errors.ChecksumError must subclass FormatError (SPEC-07 section 3)")
  232     if cls in (Exception, ValueError, BaseException) or issubclass(NotImplementedError, cls):
  233         fail("wlengine.errors.%s must be a dedicated class (SPEC-07 section 3)" % name)
  234     return cls

  237 def _after_call(label: str) -> None:
  238     if _STATE["violations"]:
  239         p = _STATE["violations"][0]

  247 def call(dotted: str, *args, allow_reference_log=False, **kw):
  248     """Call wlengine.<dotted>(*args, **kw); any exception becomes a clean failure."""
  249     fn = api(dotted)
  250     label = "wlengine." + dotted
  251     try:
  252         out = _armed(fn, *args, allow_reference_log=allow_reference_log, **kw)
  253     except NotImplementedError as e:
  254         _after_call(label)
  255         fail("%s: not implemented (%s)" % (label, e or "NotImplementedError"))
  256     except Exception as e:  # noqa: BLE001
  257         _after_call(label)
  258         fail("%s raised %s: %s" % (label, type(e).__name__, short(str(e), 200)))
  259     _after_call(label)
  260     return out
  261

  263 def call_capture(dotted: str, *args, allow_reference_log=False, **kw):
  264     """Like call() but returns (value, exception) instead of failing on an
  265     exception other than NotImplementedError or a missing API."""
  266     fn = api(dotted)
  267     label = "wlengine." + dotted
  268     try:
  269         out = _armed(fn, *args, allow_reference_log=allow_reference_log, **kw)
  270     except NotImplementedError as e:
  271         _after_call(label)
  272         fail("%s: not implemented (%s)" % (label, e or "NotImplementedError"))
  273     except Exception as e:  # noqa: BLE001
  274         _after_call(label)
  275         return None, e
  276     _after_call(label)
  277     return out, None

  280 def expect_error(err_name: str, dotted: str, *args, what: str = "", **kw) -> None:
  281     """Pass only if wlengine.<dotted> raises wlengine.errors.<err_name>."""
  282     cls = error_class(err_name)
  283     out, exc = call_capture(dotted, *args, **kw)
  284     label = "wlengine." + dotted + (" (%s)" % what if what else "")
  285     if exc is None:
  286         fail("%s: expected %s, but it returned %s" % (label, err_name, short(out)))
  287     if not isinstance(exc, cls):
  288         fail("%s: expected %s, got %s: %s" % (label, err_name, type(exc).__name__, short(str(exc), 160)))

  319 class RefRng:
  320     """SPEC-04 LR-1 and LR-2, restated for computing expected values."""
  321 
  322     def __init__(self, seed: int) -> None:
  323         self.state = seed & 0xFFFFFFFF or 0x6D2B79F5
  324 
  325     def next(self) -> int:
  326         x = self.state
  327         x ^= (x << 13) & 0xFFFFFFFF
  328         x ^= x >> 17
  329         x ^= (x << 5) & 0xFFFFFFFF
  330         self.state = x
  331         return x
  332 
  333     def roll(self, n: int) -> int:
  334         return 1 + self.next() % n
  335 
  336     def rnd(self, n: int) -> int:
  337         if n == 0:
  338             return 0
  339         if n == 1:
  340             return 1
  341         return self.roll(n)
  342 
  343     def dice(self, n: int, s: int) -> int:
  344         return sum(self.roll(s) for _ in range(n))
  345 
  346     def d100(self) -> int:
  347         return self.roll(100)

  431 def ref_pay(moneys: list[int], first_slot: int, price: int):
  432     if sum(moneys) < price:
  433         return None
  434     out = list(moneys)
  435     order = [first_slot - 1] + [i for i in range(len(moneys)) if i != first_slot - 1]
  436     left = price
  437     for i in order:
  438         take = min(out[i], left)
  439         out[i] -= take
  440         left -= take
  441         if left == 0:
  442             break
  443     return out

  446 def ref_advance_clock(clock: dict, minutes: int, seconds: int):
  447     day, minute, second = clock["day"], clock["minute"], clock["second"]
  448     total_s = second + seconds
  449     second = total_s % 60
  450     add_min = minutes + total_s // 60
  451     hours = []
  452     for _ in range(add_min):
  453         minute += 1
  454         if minute == 1440:
  455             minute = 0
  456         if minute % 60 == 0:
  457             h = minute // 60
  458             hours.append(h)
  459             if h == 0:
  460                 day += 1
  461     return {"day": day, "minute": minute, "second": second}, hours

  464 def ref_build_groups(rng: RefRng, mobs: list[dict], record: dict) -> list[dict]:
  465     counts = []
  466     for mob_index, rnd_flag, size in record["groups"]:
  467         if mob_index == 0:
  468             continue
  469         mob = mobs[mob_index - 1]
  470         s = size if size else mob["maxGroup"]
  471         counts.append((mob_index, rng.rnd(s) if rnd_flag else s))
  472     out = []
  473     for mob_index, count in counts:
  474         if count <= 0:
  475             continue
  476         mob = mobs[mob_index - 1]
  477         hp = [ref_monster_hp(rng, mob["sturdiness"]) for _ in range(count)]
  478         band = 3 if mob["weaponType"] >= 3 else 2
  479         out.append({"monster": mob_index, "count": count, "band": band, "hp": hp})
  480     return out

  492 def ref_labstate(blob: bytes) -> dict:
  493     """Independent LABSTATE.BIN reader (SPEC-04 LR-57)."""
  494     if blob[:8] != b"WLLABSV1":
  495         raise ValueError("bad magic")
  496     n = int.from_bytes(blob[0x14:0x16], "little")
  497     end = 0x16 + 6 * n
  498     if len(blob) != end + 2:
  499         raise ValueError("bad length")
  500     if sum(blob[:end]) & 0xFFFF != int.from_bytes(blob[end:end + 2], "little"):
  501         raise ValueError("bad sum")
  502     cc = [list(blob[0x16 + 6 * i:0x16 + 6 * i + 6]) for i in range(n)]
  503     return {"steps": int.from_bytes(blob[8:12], "little"), "day": int.from_bytes(blob[12:14], "little"),
  504             "seconds": blob[14], "pad": blob[15], "rng_state": int.from_bytes(blob[16:20], "little"),
  505             "cell_changes": cc}

  508 def ref_encode_labstate(state: dict) -> bytes:
  509     cc = sorted(state["cell_changes"], key=lambda c: (c[0], c[2], c[1]))
  510     b = bytearray(b"WLLABSV1")
  511     b += state["steps"].to_bytes(4, "little") + state["day"].to_bytes(2, "little")
  512     b += bytes((state["seconds"], 0)) + state["rng_state"].to_bytes(4, "little")
  513     b += len(cc).to_bytes(2, "little")
  514     for c in cc:
  515         b += bytes(c)
  516     b += (sum(b) & 0xFFFF).to_bytes(2, "little")
  517     return bytes(b)

  520 def pack4(rows: list[list[int]]) -> bytes:
  521     out = bytearray()
  522     for row in rows:
  523         for x in range(0, len(row), 2):
  524             out.append((row[x] << 4) | row[x + 1])
  525     return bytes(out)

  528 def unpack4(data: bytes, width: int) -> list[list[int]]:
  529     stride = width // 2
  530     return [[v for b in data[y * stride:(y + 1) * stride] for v in (b >> 4, b & 15)]
  531             for y in range(len(data) // stride)]
