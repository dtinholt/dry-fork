  27 def table(argnames, cases, ids=None):
  28     """Run one test function over a table of cases inside a single test.
  29 
  30     Each SPEC-04 rule is one test; its table rows are checked in order and the
  31     failure message names the first failing row only."""
  32     names = [n.strip() for n in argnames.split(",")]
  33 
  34     def deco(fn):
  35         def wrapper():
  36             for case in cases:
  37                 fn(**dict(zip(names, case if len(names) > 1 else (case,))))
  38         wrapper.__name__ = fn.__name__
  39         wrapper.__qualname__ = fn.__qualname__
  40         wrapper.__doc__ = fn.__doc__
  41         wrapper.__module__ = fn.__module__
  42         return wrapper
  43     return deco
