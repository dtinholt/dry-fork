 198|     def dispatch(self, key):
 199|         low = key.lower()
 200|         if self.pending == "char_member" and key.isdigit():
 201|             member = int(key)
 202|             if 1 <= member <= len(self.party):
 203|                 self.pending = None
 204|                 self.pending_member = member
 205|                 self.mode = "char"
 206|                 self.emit("char_open", member=member)
 207|                 return True
 208|             return False
 209|         if self.mode == "shop":
 210|             return self.shop_key(key, low)
 211|         if self.mode == "char":
 212|             return self.char_key(key, low)
 213|         if self.mode == "combat":
 214|             return self.combat_key(key, low)
 215|         handler = getattr(self, "key_" + low, None)
 216|         if handler is None:
 217|             return False
 218|         return handler() is not False

 351|     def shop_buy(self, member, iid):
 352|         rec = self.item_record(iid) or {}
 353|         price = self.buy_price(rec.get("price", 0))
 354|         m = self.party[member - 1]
 355|         m["money"] = m.get("money", 0) - price
 356|         self.emit("money", member=member, delta=-price, total=m["money"])
 357|         items = self.member_items(member)
 358|         itype = rec.get("type", 0)
 359|         if itype == 3:  # ammunition stacks
 360|             for i, entry in enumerate(items):
 361|                 if entry[0] == iid and entry[1] < 127:
 362|                     entry[1] += 1
 363|                     self.emit("buy", member=member, item=iid, price=price,
 364|                               slot=i + 1, status=entry[1])
 365|                     return
 366|         status = 1 if itype == 3 else (rec.get("capacity", 0) if itype == 5 else 0)
 367|         slot = None

 369|             if entry[0] == 0:

 500|         nx, ny = self.x + dx, self.y + dy

 502|         size = m["size"]
 503|         if not (0 <= nx < size and 0 <= ny < size):
 504|             self.emit("blocked", x=nx, y=ny)
 505|             return
 506|         if not maps.is_passable(m, nx, ny):

 142|         body += bytes(act[y])
 143|     cd_at = len(body)
 144|     body += bytes(42)
 145|     body += encode_mapinfo(S, m["info"])
 146| 
 147|     offsets = [0] * 16
 148| 
 149|     def emit_table(c: int) -> None:
 150|         recs = m["actions"].get(c)
 151|         if not recs:
 152|             return
 153|         start = len(body)
 154|         offsets[c] = start
 155|         encs = [None] + [encode_record(c, r) for r in recs]  # slot 0 is null
 156|         pos = start + 2 * len(encs)
 157|         ptrs = []
 158|         for e in encs:
 159|             if e is None:
 160|                 ptrs.append(0)
 161|             else:
 162|                 ptrs.append(pos)
 163|                 pos += len(e)
 164|         for p in ptrs:
 165|             body.extend(_u16(p))
 166|         for e in encs:
 167|             if e is not None:
 168|                 body.extend(e)
 169| 
 170|     for c in TABLE_ORDER_A:
 171|         emit_table(c)
 172|     npc_off = None
 173|     if m["npcs"]:
 174|         npc_off = len(body)
 175|         n = len(m["npcs"])
 176|         body.extend(_u16(0))
 177|         first = npc_off + 2 + 2 * n
 178|         for i in range(n):
 179|             body.extend(_u16(first + i * 0x100))
 180|         for ch in m["npcs"]:
 181|             body.extend(containers.encode_character(ch))
 182|     for c in TABLE_ORDER_B:
 183|         emit_table(c)
 184|     specials_off = 0  # lab rule: no special-action table
 185|     names_off = len(body)
 186|     for mob in m["mobs"]:
 187|         body.extend(mob["name"] + b"\x00")
 188|     data_off = len(body)
 189|     for mob in m["mobs"]:
 190|         body.extend(encode_mob(mob))
 191|     strings_off = len(body)
 192|     if npc_off is None:
 193|         npc_off = strings_off  # lab rule: no NPCs -> npcOffset = stringsOffset
 194|     body.extend(strings.encode_table(m["dictionary"], m["strings"]))
 195| 
 196|     cd = _u16(strings_off) + _u16(names_off) + _u16(data_off)
 197|     for o in offsets:
 198|         cd += _u16(o)
 199|     cd += _u16(specials_off) + _u16(npc_off)
 200|     body[cd_at:cd_at + 42] = cd
 201| 
 202|     prefix = bytes(body[:strings_off])
 203|     if not codecs.rotating_xor_prefix_is_unambiguous(prefix):
 204|         raise FormatError("scenario: map %r has an ambiguous encrypted prefix"
 205|                           % m.get("name", "?"))
 206|     enc = codecs.rotating_xor_encode(prefix)
 207|     data_section = enc + bytes(body[strings_off:]) + b"\x00"  # final byte: unknown, lab 0
 208|     tile_map = bytes(v for row in img for v in row)
 209|     tsec = _u32(S * S) + _u32(0) + huffman.encode(tile_map)
 210|     header = b"msq" + bytes(((0x30 + disk) & 0xFF,))
 211|     block = header + data_section + tsec
 212|     tmo = len(header) + len(data_section)
 213|     facts = {"bodySize": len(body), "encryptedPrefixLength": strings_off,
 214|              "centralDirectory": {"stringsOffset": strings_off, "mobNamesOffset": names_off,
 215|                                   "mobDataOffset": data_off, "actionClassOffsets": offsets,
 216|                                   "specialsOffset": specials_off, "npcOffset": npc_off}}
 217|     return block, tmo, facts

 339| def _encode_labtables(t: dict) -> bytes:
 340|     out = bytearray(containers.LABTABLES_SIZE)
 341|     out[0:8] = containers.LABTABLES_MAGIC
 342|     for k, (off, fmt, n) in containers.LT.items():
 343|         vals = t[k] if isinstance(t[k], list) else [t[k]]
 344|         if len(vals) != n:
 345|             raise FormatError("LABTABLES: %s has %d values, expected %d" % (k, len(vals), n))
 346|         w = containers._WIDTH[fmt]
 347|         for i, v in enumerate(vals):
 348|             out[off + i * w:off + (i + 1) * w] = (v & ((1 << (8 * w)) - 1)).to_bytes(w, "little")
 349|     return bytes(out)

 423| def _planes_encode(pix: list, bytes_per_row: int) -> bytes:
 424|     """Four 1-bit planes (blue, green, red, intensity), MSB = leftmost pixel."""
 425|     out = bytearray()
 426|     for bit in range(4):
 427|         for row in pix:
 428|             for bx in range(bytes_per_row):
 429|                 v = 0
 430|                 for i in range(8):
 431|                     v |= ((row[bx * 8 + i] >> bit) & 1) << (7 - i)
 432|                 out.append(v)
 433|     return bytes(out)
 434| 
 435| 
 436| def _mask_encode(mask: list) -> bytes:
 437|     out = bytearray()
 438|     for row in mask:
 439|         for bx in range(2):
 440|             v = 0
 441|             for i in range(8):
 442|                 v |= (row[bx * 8 + i] & 1) << (7 - i)
 443|             out.append(v)
 444|     return bytes(out)
 445| 
 446| 
 447| def _cursor_encode(pix: list, vis: list) -> bytes:
 448|     out = bytearray()
 449|     for bit in range(4):
 450|         for y in range(16):
 451|             def byte(src, x0, fn):
 452|                 v = 0
 453|                 for i in range(8):
 454|                     v |= (fn(src[y][x0 + i]) & 1) << (7 - i)
 455|                 return v
 456|             out += bytes((byte(vis, 8, int), byte(vis, 0, int),
 457|                           byte(pix, 8, lambda p: p >> bit), byte(pix, 0, lambda p: p >> bit)))
 458|     return bytes(out)

 461| def _portrait_anim_encode(scripts: list, updates: list) -> bytes:
 462|     sb = bytearray()
 463|     for sc in scripts:
 464|         for d, u in sc:
 465|             sb += bytes((d, u))
 466|         sb.append(0xFF)
 467|     ub = bytearray()
 468|     for upd in updates:
 469|         for off, data in upd:
 470|             ub += _u16(((len(data) - 1) << 12) | off) + data
 471|         ub += _u16(0xFFFF)
 472|     return _u16(len(sb)) + bytes(sb) + _u16(len(ub)) + bytes(ub)
 473|

 475| def _ending_anim_encode(updates: list) -> bytes:
 476|     st = bytearray()
 477|     for delay, patches in updates:
 478|         st += _u16(delay)
 479|         for off, data in patches:
 480|             st += _u16(off) + data
 481|         st += _u16(0xFFFF)
 482|     st += _u16(0xFFFF)
 483|     return _u16(len(st)) + bytes(st) + _u16(0)

 486| def _placeholder_tile(t: int) -> list:
 487|     """Tile ``t``: fill ``t mod 16``, one-pixel border ``(t + 8) mod 16``."""
 488|     pix = [[t % 16 for _ in range(16)] for _ in range(16)]
 489|     border = (t + 8) % 16
 490|     for x in range(16):
 491|         pix[0][x] = border
 492|         pix[15][x] = border
 493|     for y in range(16):
 494|         pix[y][0] = border
 495|         pix[y][15] = border
 496|     return pix

 500|     """Portrait ``p``: fill ``4 + p``, centred 48 x 42 rectangle of colour 15."""
 501|     pix = [[4 + p for _ in range(96)] for _ in range(84)]
 502|     for y in range(21, 63):
 503|         for x in range(24, 72):
 504|             pix[y][x] = 15

 529| def _build_sprites():
 530|     sprites, masks = bytearray(), bytearray()
 531|     for n in range(10):

 541|     out = bytearray()

 543|         pix = [[((g >> (x % 5)) & 1) * ((g + y) % 15 + 1) for x in range(8)]
 544|                for y in range(8)]

 552|         pix = [[(x + y + k) % 16 for x in range(16)] for y in range(16)]

 586| def _call(mod, names, *args, **kwargs):
 587|     for n in names:
 588|         fn = getattr(mod, n, None)

 590|             return fn(*args, **kwargs)

 645|     def _load_maps(self):
 646|         counts = self.tables["mapCounts"]
 647|         for loc in range(len(self.tables["locationMapping"])):
 648|             code = self.tables["locationMapping"][loc]
 649|             if code == 0:
 650|                 continue
 651|             disk = code >> 6
 652|             if disk == 2 and counts[0] == 0:
 653|                 continue
 654|             if disk == 1 and counts[1] == 0:
 655|                 continue
 656|             try:
 657|                 self.maps[loc] = maps.load_map(self.dir, loc)
 658|             except FormatError:
 659|                 continue

 694|                 "max_hp": c["maxCon"], "hp": c["curCon"], "cond": cond,

 786|     def item(self, item_id):
 787|         """Item id N is record N - 1 of shop list 0 (LR-47)."""
 788|         if item_id < 1:
 789|             return None

 792|             return None

 795|     def skill(self, skill_id):
 796|         """``(min_iq, cost, attr_offset)`` for skill id 1 to 35 (LR-6)."""
 797|         if skill_id < 1 or skill_id >= len(self.skill_table):
 798|             return (0, 0, 0)
 799|         b0, b1 = self.skill_table[skill_id]
 800|         return (b0 >> 3, b0 & 7, b1)

 833| def _compile(scenario_dir: str, out_dir: str) -> None:
 834|     """Compile the scenario in ``scenario_dir`` into ``out_dir`` (SPEC-05 13)."""
 835|     scen = _load_json(os.path.join(scenario_dir, "scenario.json"))
 836|     items = _load_json(os.path.join(scenario_dir, "items.json"))["items"]
 837|     catalog = {it["id"]: it for it in items}
 838|     ranks = scen["labstr_tables"]["3"]["strings"]
 839|     os.makedirs(out_dir, exist_ok=True)
 840| 
 841|     game = bytearray()
 842|     map_facts = []
 843|     for loc in scen["locations"]:
 844|         m = _load_json(os.path.join(scenario_dir, loc["file"]))
 845|         mm = _map_from_json(m, loc["size"], catalog, ranks)
 846|         mm["name"] = loc["name"]
 847|         block, tmo, facts = build_map(mm, 0)
 848|         map_facts.append((len(game), tmo, facts, loc))
 849|         game += block
 850| 
 851|     sg_off = len(game)
 852|     game += _build_savegame(scenario_dir, scen, catalog, ranks)
 853| 
 854|     shops = _load_json(os.path.join(scenario_dir, "shops.json"))["shops"]
 855|     for sh in shops:
 856|         game += containers.encode_shop_list(_shop_items(sh, catalog), 0)
 857|     _write(out_dir, "GAME1", bytes(game))

 859|     allhtds = _build_allhtds(scenario_dir)

 870|     _write(out_dir, "END.CPA", _build_ending())

 872|     _write(out_dir, "LABTABLES.BIN",
 873|            _build_labtables(map_facts, sg_off, len(allhtds), portrait_offsets))
