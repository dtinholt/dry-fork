   1| """LABSTATE.BIN (SPEC-04 LR-57).
   2| 
   3| Little-endian file: magic ``WLLABSV1``, step counter, day, clock seconds, a
   4| pad byte, the generator state, the cell-change count and the cell changes
   5| sorted by (location, y, x), then a u16 sum of all earlier bytes.
   6| """
   7| 
   8| from __future__ import annotations
   9| 
  10| from .errors import FormatError
  11| 
  12| MAGIC = b"WLLABSV1"
  13| 
  14| 
  15| def encode_labstate(state: dict) -> bytes:
  16|     """LR-57: cell changes written sorted by (location, y, x)."""
  17|     cc = sorted(state.get("cell_changes") or [], key=lambda c: (c[0], c[2], c[1]))
  18|     b = bytearray(MAGIC)
  19|     b += int(state.get("steps", 0)).to_bytes(4, "little")
  20|     b += int(state.get("day", 1)).to_bytes(2, "little")
  21|     b += bytes((int(state.get("seconds", 0)) & 0xFF, 0))
  22|     b += int(state.get("rng_state", 0)).to_bytes(4, "little")
  23|     b += len(cc).to_bytes(2, "little")
  24|     for c in cc:
  25|         b += bytes(int(v) & 0xFF for v in c)
  26|     b += (sum(b) & 0xFFFF).to_bytes(2, "little")
  27|     return bytes(b)
  28| 
  29| 
  30| def decode_labstate(data: bytes) -> dict:
  31|     """LR-57; ``FormatError`` on a wrong magic or sum."""
  32|     if len(data) < 0x16 + 2:
  33|         raise FormatError("LABSTATE: %d bytes, too short" % len(data))
  34|     if data[:8] != MAGIC:
  35|         raise FormatError("LABSTATE: bad magic %r" % (bytes(data[:8]),))
  36|     n = int.from_bytes(data[0x14:0x16], "little")
  37|     end = 0x16 + 6 * n
  38|     if len(data) != end + 2:
  39|         raise FormatError("LABSTATE: %d bytes, expected %d" % (len(data), end + 2))
  40|     if sum(data[:end]) & 0xFFFF != int.from_bytes(data[end:end + 2], "little"):
  41|         raise FormatError("LABSTATE: bad checksum")
  42|     cc = [list(data[0x16 + 6 * i:0x16 + 6 * i + 6]) for i in range(n)]
  43|     return {"steps": int.from_bytes(data[8:12], "little"),
  44|             "day": int.from_bytes(data[12:14], "little"),
  45|             "seconds": data[14], "pad": data[15],
  46|             "rng_state": int.from_bytes(data[16:20], "little"),
  47|             "cell_changes": cc}
