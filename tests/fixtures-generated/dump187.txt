  195 class Runner:
  196     """Cached headless runs on fresh copies of the compiled data."""
  197 
  198     def __init__(self, compiled: Compiled, tmp_path_factory) -> None:
  199         self.compiled = compiled
  200         self.tpf = tmp_path_factory
  201         self.cache: dict = {}
  202 
  203     def run(self, key: str, script: dict, seed: int | None = None, data_dir: Path | None = None):
  204         """Return (events, data_dir). Fails the test cleanly on any engine problem."""
  205         ck = (key, seed, str(data_dir) if data_dir else None)
  206         if ck in self.cache:
  207             ev, dd, err = self.cache[ck]
  208             if err:
  209                 W.fail(err)
  210             return ev, dd
  211         dd = data_dir or self.compiled.copy(self.tpf)
  212         err = None
  213         ev = None
  214         try:
  215             ev = W.call("headless.run", str(dd), json.loads(json.dumps(script)), seed)
  216             if not isinstance(ev, list) or not all(isinstance(e, dict) for e in ev):
  217                 err = "wlengine.headless.run must return a list of event dicts, got %s" % type(ev).__name__
  218         except pytest.fail.Exception as e:
  219             err = str(e)
  220         self.cache[ck] = (ev, dd, err)
  221         if err:
  222             W.fail(err)
  223         return ev, dd
