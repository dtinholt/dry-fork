   71 def first_diff(got, want, path: str = ""):
   72     """Return (path, got_part, want_part) of the first difference, or None."""
   73     if isinstance(want, dict):
   74         if not isinstance(got, dict):
   75             return (path or "<root>", got, "a dict")
   76         for k in want:
   77             if k not in got:
   78                 return ("%s.%s" % (path, k) if path else k, "<missing key>", want[k])
   79             d = first_diff(got[k], want[k], "%s.%s" % (path, k) if path else str(k))
   80             if d:

  492 def ref_labstate(blob: bytes) -> dict:
  493     """Independent LABSTATE.BIN reader (SPEC-04 LR-57)."""
  494     if blob[:8] != b"WLLABSV1":
  495         raise ValueError("bad magic")
  496     n = int.from_bytes(blob[0x14:0x16], "little")
  497     end = 0x16 + 6 * n

  508 def ref_encode_labstate(state: dict) -> bytes:
  509     cc = sorted(state["cell_changes"], key=lambda c: (c[0], c[2], c[1]))
  510     b = bytearray(b"WLLABSV1")
  511     b += state["steps"].to_bytes(4, "little") + state["day"].to_bytes(2, "little")
  512     b += bytes((state["seconds"], 0)) + state["rng_state"].to_bytes(4, "little")

  600     def _until_met(self, until):
  601         if isinstance(until, str):
  602             if until.startswith("mode:"):

  604             if until.startswith("cell:"):

  441 def prefix(walkthrough, upto: str) -> list:
  442     out = []
  443     for item in walkthrough["script"]:
  444         out.append(item)
  445         if isinstance(item, dict) and item.get("checkpoint") == upto:
  446             return out
  447     raise AssertionError("no checkpoint %s in walkthrough.json" % upto)

  450 def opening_script(walkthrough) -> list:
  451     out = []
  452     for item in walkthrough["script"]:

  455         out.append(item)
  456     raise AssertionError("walkthrough.json has no repeat_until macro")

  459 def ckpt(events, name):
  460     for e in events:
  461         if e.get("type") == "checkpoint" and e.get("name") == name:
  462             st = e.get("state")
  463             if not isinstance(st, dict):
  464                 fail("checkpoint %s has no state object" % name)
  465             return st
  466     fail("the engine log has no checkpoint %r" % name)

  473 def between(events, a, b):
  474     ia = next((i for i, e in enumerate(events) if e.get("type") == "checkpoint" and e.get("name") == a), None)
  475     ib = next((i for i, e in enumerate(events) if e.get("type") == "checkpoint" and e.get("name") == b), None)
  476     if ia is None or ib is None:
  477         fail("the engine log lacks checkpoint %s or %s" % (a, b))
  478     return [strip(e) for e in events[ia + 1:ib]]

   79 def encode_record(cls: int, r: dict) -> bytes:

  183         emit_table(c)
  184     specials_off = 0  # lab rule: no special-action table
  185     names_off = len(body)
  186     for mob in m["mobs"]:
  187         body.extend(mob["name"] + b"\x00")
  188     data_off = len(body)
  189     for mob in m["mobs"]:
  190         body.extend(encode_mob(mob))
  191     strings_off = len(body)
  192     if npc_off is None:
  193         npc_off = strings_off  # lab rule: no NPCs -> npcOffset = stringsOffset
  194     body.extend(strings.encode_table(m["dictionary"], m["strings"]))
  195 
  196     cd = _u16(strings_off) + _u16(names_off) + _u16(data_off)
  197     for o in offsets:
  198         cd += _u16(o)
  199     cd += _u16(specials_off) + _u16(npc_off)
  200     body[cd_at:cd_at + 42] = cd
  201 
  202     prefix = bytes(body[:strings_off])
  203     if not codecs.rotating_xor_prefix_is_unambiguous(prefix):
  204         raise FormatError("scenario: map %r has an ambiguous encrypted prefix"
  205                           % m.get("name", "?"))
  206     enc = codecs.rotating_xor_encode(prefix)
  207     data_section = enc + bytes(body[strings_off:]) + b"\x00"  # final byte: unknown, lab 0
  208     tile_map = bytes(v for row in img for v in row)
  209     tsec = _u32(S * S) + _u32(0) + huffman.encode(tile_map)
  210     header = b"msq" + bytes(((0x30 + disk) & 0xFF,))
  211     block = header + data_section + tsec
  212     tmo = len(header) + len(data_section)
  213     facts = {"bodySize": len(body), "encryptedPrefixLength": strings_off,
  214              "centralDirectory": {"stringsOffset": strings_off, "mobNamesOffset": names_off,
  215                                   "mobDataOffset": data_off, "actionClassOffsets": offsets,
  216                                   "specialsOffset": specials_off, "npcOffset": npc_off}}
  217     return block, tmo, facts
  218

  339 def _encode_labtables(t: dict) -> bytes:
  340     out = bytearray(containers.LABTABLES_SIZE)
  341     out[0:8] = containers.LABTABLES_MAGIC
  342     for k, (off, fmt, n) in containers.LT.items():
  343         vals = t[k] if isinstance(t[k], list) else [t[k]]
  344         if len(vals) != n:
  345             raise FormatError("LABTABLES: %s has %d values, expected %d" % (k, len(vals), n))
  346         w = containers._WIDTH[fmt]
  347         for i, v in enumerate(vals):
  348             out[off + i * w:off + (i + 1) * w] = (v & ((1 << (8 * w)) - 1)).to_bytes(w, "little")
  349     return bytes(out)

  381 def _labstr_file(tables: list) -> bytes:
  382     n = len(tables)
  383     head = strings.LABSTR_MAGIC + _u16(n) + _u16(0)
  384     pos = len(head) + 8 * n
  385     offs, sizes = b"", b""
  386     for t in tables:
  387         offs += _u32(pos)
  388         sizes += _u32(len(t))
  389         pos += len(t)
  390     return head + offs + sizes + b"".join(tables)

  423 def _planes_encode(pix: list, bytes_per_row: int) -> bytes:
  424     """Four 1-bit planes (blue, green, red, intensity), MSB = leftmost pixel."""
  425     out = bytearray()
  426     for bit in range(4):
  427         for row in pix:
  428             for bx in range(bytes_per_row):
  429                 v = 0
  430                 for i in range(8):
  431                     v |= ((row[bx * 8 + i] >> bit) & 1) << (7 - i)
  432                 out.append(v)
  433     return bytes(out)
  434 
  435 
  436 def _mask_encode(mask: list) -> bytes:
  437     out = bytearray()
  438     for row in mask:
  439         for bx in range(2):
  440             v = 0
  441             for i in range(8):
  442                 v |= (row[bx * 8 + i] & 1) << (7 - i)
  443             out.append(v)
  444     return bytes(out)
  445 
  446 
  447 def _cursor_encode(pix: list, vis: list) -> bytes:
  448     out = bytearray()
  449     for bit in range(4):
  450         for y in range(16):
  451             def byte(src, x0, fn):
  452                 v = 0
  453                 for i in range(8):
  454                     v |= (fn(src[y][x0 + i]) & 1) << (7 - i)
  455                 return v
  456             out += bytes((byte(vis, 8, int), byte(vis, 0, int),
  457                           byte(pix, 8, lambda p: p >> bit), byte(pix, 0, lambda p: p >> bit)))
  458     return bytes(out)

  461 def _portrait_anim_encode(scripts: list, updates: list) -> bytes:
  462     sb = bytearray()
  463     for sc in scripts:
  464         for d, u in sc:
  465             sb += bytes((d, u))
  466         sb.append(0xFF)
  467     ub = bytearray()
  468     for upd in updates:
  469         for off, data in upd:
  470             ub += _u16(((len(data) - 1) << 12) | off) + data
  471         ub += _u16(0xFFFF)
  472     return _u16(len(sb)) + bytes(sb) + _u16(len(ub)) + bytes(ub)
  473 
  474 
  475 def _ending_anim_encode(updates: list) -> bytes:
  476     st = bytearray()
  477     for delay, patches in updates:
  478         st += _u16(delay)
  479         for off, data in patches:
  480             st += _u16(off) + data
  481         st += _u16(0xFFFF)
  482     st += _u16(0xFFFF)
  483     return _u16(len(st)) + bytes(st) + _u16(0)

  486 def _placeholder_tile(t: int) -> list:
  487     """Tile ``t``: fill ``t mod 16``, one-pixel border ``(t + 8) mod 16``."""
  488     pix = [[t % 16 for _ in range(16)] for _ in range(16)]
  489     border = (t + 8) % 16
  490     for x in range(16):
  491         pix[0][x] = border
  492         pix[15][x] = border
  493     for y in range(16):
  494         pix[y][0] = border
  495         pix[y][15] = border
  496     return pix

  499 def _placeholder_portrait(p: int) -> list:
  500     """Portrait ``p``: fill ``4 + p``, centred 48 x 42 rectangle of colour 15."""
  501     pix = [[4 + p for _ in range(96)] for _ in range(84)]
  502     for y in range(21, 63):
  503         for x in range(24, 72):
  504             pix[y][x] = 15
  505     return pix

  515 def _build_allpics(scenario_dir: str):

  529 def _build_sprites():
  530     sprites, masks = bytearray(), bytearray()
  531     for n in range(10):
  532         pix = [[(n + x // 4 + y // 4) % 16 for x in range(16)] for y in range(16)]
  533         msk = [[1 if (x - 7.5) ** 2 + (y - 7.5) ** 2 > 56 else 0
  534                 for x in range(16)] for y in range(16)]
  535         sprites += _planes_encode(pix, 2)
  536         masks += _mask_encode(msk)
  537     return bytes(sprites), bytes(masks)

  540 def _build_font() -> bytes:
  541     out = bytearray()
  542     for g in range(32):
  543         pix = [[((g >> (x % 5)) & 1) * ((g + y) % 15 + 1) for x in range(8)]
  544                for y in range(8)]
  545         out += _planes_encode(pix, 1)
  546     return bytes(out)

  549 def _build_cursors() -> bytes:
  550     out = bytearray()
  551     for k in range(2):
  552         pix = [[(x + y + k) % 16 for x in range(16)] for y in range(16)]
  553         vis = [[1 if x <= y + k * 3 else 0 for x in range(16)] for y in range(16)]
  554         out += _cursor_encode(pix, vis)
  555     return bytes(out)

  558 def _build_title() -> bytes:

  563 def _build_ending() -> bytes:
  564     base = graphics.pack4([[((x + y) // 24) % 16 for x in range(288)] for y in range(128)])
  565     updates = [(5, [(41, bytes.fromhex("11223344")), (40 * 100 + 35, bytes.fromhex("FFEEDDCC"))]),
  566                (9, [(0, bytes.fromhex("0F0F0F0F"))])]
  567     anim = _ending_anim_encode(updates)
  568     return (huffman.encode_msq(codecs.vertical_xor_encode(base, TITLE_STRIDE), 0)
  569             + huffman.encode_msq(anim, 0, ENDING_SIGNATURE))

  586 def _call(mod, names, *args, **kwargs):
  587     for n in names:
  588         fn = getattr(mod, n, None)
  589         if callable(fn):
  590             return fn(*args, **kwargs)
  591     raise FormatError("scenario: %s has none of %r" % (mod.__name__, names))

  594 class Scenario:
  595     """A compiled lab data set, read back through the engine's own loaders."""
  596 
  597     def __init__(self, data_dir):
  598         self.dir = data_dir
  599         self.maps = {}
  600         self.hooks = []
  601         self.paragraphs = {}
  602         self.skills = {}
  603         self.shops = {}
  604         self.savegame = None
  605         self.start = {"location": 0, "x": 0, "y": 0, "hour": 8, "minute": 0,
  606                       "day": 1, "paragraph": 1}
  607         self._load()
  608 
  609     # -- raw file access ---------------------------------------------------
  610 
  611     def _read(self, name):
  612         path = os.path.join(self.dir, name)
  613         if not os.path.exists(path):
  614             return None
  615         with open(path, "rb") as fh:
  616             return fh.read()
  617 
  618     def _json(self, name):
  619         path = os.path.join(self.dir, name)
  620         if not os.path.exists(path):
  621             return None
  622         return _load_json(path)
  623 
  624     # -- loading -----------------------------------------------------------
  625 
  626     def _load(self):
  627         self.tables = containers.read_tables(os.path.join(self.dir, "LABTABLES.BIN"))
  628         self.game1 = self._read("GAME1")
  629         self._load_maps()
  630         self._load_savegame()
  631         self._load_shops()
  632         labstr = os.path.join(self.dir, "LABSTR.BIN")
  633         if os.path.exists(labstr):
  634             self.labstr = strings.read_labstr(labstr)
  635         skills = os.path.join(self.dir, "LABSKILL.BIN")
  636         if os.path.exists(skills):
  637             self.skill_table = rules.read_skill_table(skills)
  638         hooks = self._json("LABHOOKS.JSON")
  639         if hooks is not None:
  640             self.hooks = hooks if isinstance(hooks, list) else hooks.get("hooks", [])
  641         paras = self._json("PARAGRAPHS.JSON")
  642         if paras is not None:
  643             self.paragraphs = paras.get("paragraphs", paras) if isinstance(paras, dict) else paras
  644 
  645     def _load_maps(self):
  646         counts = self.tables["mapCounts"]
  647         for loc in range(len(self.tables["locationMapping"])):
  648             code = self.tables["locationMapping"][loc]
  649             if code == 0:
  650                 continue
  651             disk = code >> 6
  652             if disk == 2 and counts[0] == 0:
  653                 continue
  654             if disk == 1 and counts[1] == 0:
  655                 continue
  656             try:
  657                 self.maps[loc] = maps.load_map(self.dir, loc)
  658             except FormatError:
  659                 continue
  660 
  661     def _load_savegame(self):
  662         offs = self.tables["savegameOffsets"]
  663         if not offs or offs[0] == 0xFFFFFFFF:
  664             return
  665         self.savegame = containers.decode_savegame(self.game1, offs[0])
  666         sg = self.savegame
  667         g0 = sg["groups"][0]

    2 
    3 Little-endian file: magic ``WLLABSV1``, step counter, day, clock seconds, a
    4 pad byte, the generator state, the cell-change count and the cell changes
    5 sorted by (location, y, x), then a u16 sum of all earlier bytes.
    6 """

   12 MAGIC = b"WLLABSV1"

   15 def encode_labstate(state: dict) -> bytes:
   16     """LR-57: cell changes written sorted by (location, y, x)."""
   17     cc = sorted(state.get("cell_changes") or [], key=lambda c: (c[0], c[2], c[1]))
   18     b = bytearray(MAGIC)
   19     b += int(state.get("steps", 0)).to_bytes(4, "little")
   20     b += int(state.get("day", 1)).to_bytes(2, "little")
   21     b += bytes((int(state.get("seconds", 0)) & 0xFF, 0))
   22     b += int(state.get("rng_state", 0)).to_bytes(4, "little")
   23     b += len(cc).to_bytes(2, "little")
   24     for c in cc:
   25         b += bytes(int(v) & 0xFF for v in c)
   26     b += (sum(b) & 0xFFFF).to_bytes(2, "little")
   27     return bytes(b)
   28

   30 def decode_labstate(data: bytes) -> dict:
   31     """LR-57; ``FormatError`` on a wrong magic or sum."""
   32     if len(data) < 0x16 + 2:
   33         raise FormatError("LABSTATE: %d bytes, too short" % len(data))
   34     if data[:8] != MAGIC:
   35         raise FormatError("LABSTATE: bad magic %r" % (bytes(data[:8]),))
   36     n = int.from_bytes(data[0x14:0x16], "little")
   37     end = 0x16 + 6 * n
   38     if len(data) != end + 2:
   39         raise FormatError("LABSTATE: %d bytes, expected %d" % (len(data), end + 2))
   40     if sum(data[:end]) & 0xFFFF != int.from_bytes(data[end:end + 2], "little"):
   41         raise FormatError("LABSTATE: bad checksum")
   42     cc = [list(data[0x16 + 6 * i:0x16 + 6 * i + 6]) for i in range(n)]
   43     return {"steps": int.from_bytes(data[8:12], "little"),
   44             "day": int.from_bytes(data[12:14], "little"),
   45             "seconds": data[14], "pad": data[15],
   46             "rng_state": int.from_bytes(data[16:20], "little"),
   47             "cell_changes": cc}
